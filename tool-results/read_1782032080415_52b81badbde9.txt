    69→// ==================== AUTH SCREEN ====================
    70→function AuthScreen() {
    71→  const { user, setUser, setPage, addToast } = useAppStore();
    72→  // Referral code from URL (lazy init before useState)
    73→  const { prefilledReferral, initialMode } = (() => {
    74→    if (typeof window === 'undefined') return { prefilledReferral: '', initialMode: 'login' as const };
    75→    try {
    76→      const params = new URLSearchParams(window.location.search);
    77→      const ref = params.get('ref');
    78→      if (ref) return { prefilledReferral: ref.toUpperCase(), initialMode: 'register' as const };
    79→    } catch { /* */ }
    80→    return { prefilledReferral: '', initialMode: 'login' as const };
    81→  })();
    82→  const [mode, setMode] = useState<'login' | 'register'>(initialMode);
    83→  const [loading, setLoading] = useState(false);
    84→  const [showPw, setShowPw] = useState({ l: false, r: false, r2: false });
    85→  const [errors, setErrors] = useState<Record<string, string>>({});
    86→  // OTP state (registration only)
    87→  const [otpStep, setOtpStep] = useState(false);
    88→  const [otpEmail, setOtpEmail] = useState('');
    89→  const [otpCode, setOtpCode] = useState('');
    90→  const [otpLoading, setOtpLoading] = useState(false);
    91→  const [simCode, setSimCode] = useState('');
    92→
    93→  if (user) return null;
    94→
    95→  const handleLogin = async (e: React.FormEvent) => {
    96→    e.preventDefault();
    97→    setLoading(true);
    98→    try {
    99→      const fd = new FormData(e.target as HTMLFormElement);
   100→      const email = fd.get('email') as string;
   101→      const password = fd.get('password') as string;
   102→      const res = await fetch('/api/auth/login', { method: 'POST', body: JSON.stringify({ email, password }), headers: { 'Content-Type': 'application/json' } });
   103→      const data = await res.json();
   104→      if (data.success) {
   105→        setUser(data.user); addToast('Bienvenue, ' + data.user.name, 'success'); setPage('home');
   106→      } else {
   107→        addToast(data.error, 'error');
   108→      }
   109→    } catch { addToast('Erreur réseau', 'error'); }
   110→    setLoading(false);
   111→  };
   112→
   113→  const handleOtpVerify = async () => {
   114→    if (!otpCode || otpCode.length < 6) { addToast('Entrez le code à 6 chiffres', 'error'); return; }
   115→    setOtpLoading(true);
   116→    try {
   117→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'verify', email: otpEmail, code: otpCode, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   118→      const data = await res.json();
   119→      if (data.success && data.user) {
   120→        setUser(data.user);
   121→        addToast('Email vérifié ! Bienvenue, ' + data.user.name, 'success');
   122→        setPage('home');
   123→      } else {
   124→        addToast(data.error || 'Code invalide', 'error');
   125→      }
   126→    } catch { addToast('Erreur réseau', 'error'); }
   127→    setOtpLoading(false);
   128→  };
   129→
   130→  const handleResendOtp = async () => {
   131→    setOtpLoading(true);
   132→    try {
   133→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'send', email: otpEmail, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   134→      const data = await res.json();
   135→      if (data.success) {
   136→        if (data.plain_code) {
   137→          setSimCode(data.plain_code);
   138→          addToast('Code: ' + data.plain_code, 'info');
   139→        } else {
   140→          addToast('Code renvoyé à ' + otpEmail, 'success');
   141→        }
   142→      } else { addToast(data.error, 'error'); }
   143→    } catch { addToast('Erreur réseau', 'error'); }
   144→    setOtpLoading(false);
   145→  };
   146→
   147→  const handleRegister = async (e: React.FormEvent) => {
   148→    e.preventDefault();
   149→    const fd = new FormData(e.target as HTMLFormElement);
   150→    const name = (fd.get('name') as string)?.trim() || '';
   151→    const email = (fd.get('email') as string)?.trim() || '';
   152→    const password = fd.get('password') as string || '';
   153→    const password2 = fd.get('password2') as string || '';
   154→    const referralCode = (fd.get('referralCode') as string)?.trim().toUpperCase() || '';
   155→    const errs: Record<string, string> = {};
   156→    if (name.length < 2) errs.name = 'Min. 2 caractères';
   157→    if (password.length < 6) errs.password = 'Min. 6 caractères';
   158→    if (password !== password2) errs.password2 = 'Ne correspond pas';
   159→    if (Object.keys(errs).length) { setErrors(errs); return; }
   160→    setErrors({});
   161→    setLoading(true);
   162→    try {
   163→      const res = await fetch('/api/auth/register', { method: 'POST', body: JSON.stringify({ name, email, password, password2, referralCode }), headers: { 'Content-Type': 'application/json' } });
   164→      const data = await res.json();
   165→      if (data.success && data.requires_verification) {
   166→        // Account created — must verify email via OTP
   167→        setOtpEmail(email);
   168→        setOtpStep(true);
   169→        if (data.plain_code) {
   170→          setSimCode(data.plain_code);
   171→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   172→        } else {
   173→          addToast('Code de vérification envoyé à ' + email, 'success');
   174→        }
   175→      } else if (data.success) {
   176→        setUser(data.user); addToast('Compte créé !', 'success'); setPage('home');
   177→      } else {
   178→        addToast(data.error, 'error');
   179→      }
   180→    } catch { addToast('Erreur réseau', 'error'); }
   181→    setLoading(false);
   182→  };
   183→
   184→  return (
   185→    <section className="absolute inset-0 bg-gradient-to-br from-[#F0FDF4] via-[#F8F9FA] to-[#ECFDF5] flex flex-col items-center justify-center z-[200]">
   186→      {/* Floating decorative orbs */}
   187→      <div className="absolute w-[200px] h-[200px] rounded-full bg-[rgba(34,197,94,0.06)] blur-[80px] top-[15%] left-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite' }} />
   188→      <div className="absolute w-[160px] h-[160px] rounded-full bg-[rgba(20,184,166,0.05)] blur-[80px] bottom-[15%] right-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite 3s reverse' }} />
   189→      <div className="absolute w-[120px] h-[120px] rounded-full bg-[rgba(245,158,11,0.04)] blur-[80px] top-[50%] right-[25%]" style={{ animation: 'orbFloat 7s ease-in-out infinite 1.5s' }} />
   190→
   191→      <div className="w-full max-w-[330px] text-center px-5 relative z-[1]">
   192→        <LogoImg className="w-[100px] h-[100px] mx-auto mb-4" style={{ filter: 'drop-shadow(0 8px 32px rgba(34,197,94,0.25))' }} />
   193→        <h1 className="text-[1.8rem] font-black mb-1 bg-gradient-to-r from-[#22C55E] to-[#16A34A] bg-clip-text text-transparent tracking-[2px]">BE RICH</h1>
   194→
   195→        {/* OTP Verification Step — Registration only */}
   196→        {otpStep ? (
   197→          <>
   198→            <div className="w-14 h-14 mx-auto rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mb-4 mt-4">
   199→              <i className="fas fa-shield-alt text-[#22C55E] text-[1.3rem]"></i>
   200→            </div>
   201→            <h2 className="text-[1rem] font-bold text-[#1F2937] mb-1">Vérification email</h2>
   202→            <p className="text-[rgba(0,0,0,0.45)] text-[0.72rem] mb-5">
   203→              Un code a été envoyé à <strong className="text-[#1F2937]">{otpEmail}</strong> pour vérifier votre adresse email
   204→            </p>
   205→
   206→            {/* Simulation mode code display */}
   207→            {simCode && (
   208→              <div className="mb-4 p-4 bg-[rgba(245,158,11,0.12)] border-2 border-[rgba(245,158,11,0.4)] rounded-xl">
   209→                <div className="flex items-center gap-1.5 mb-2">
   210→                  <i className="fas fa-key text-[#F59E0B] text-[0.7rem]"></i>
   211→                  <div className="text-[0.72rem] text-[#D97706] font-bold">Votre code de vérification</div>
   212→                </div>
   213→                <div className="text-[2rem] font-black text-[#F59E0B] tracking-[8px] font-mono text-center py-2">{simCode}</div>
   214→                <div className="flex gap-2 mt-2">
   215→                  <button
   216→                    onClick={() => { setOtpCode(simCode); addToast('Code rempli automatiquement', 'success'); }}
   217→                    className="flex-1 py-2 rounded-lg bg-[#F59E0B] text-white text-[0.75rem] font-bold border-none cursor-pointer flex items-center justify-center gap-1"
   218→                  >
   219→                    <i className="fas fa-magic mr-1"></i>Remplir automatiquement
   220→                  </button>
   221→                  <button
   222→                    onClick={() => { try { navigator.clipboard.writeText(simCode); } catch { /* */ } addToast('Code copié', 'success'); }}
   223→                    className="py-2 px-3 rounded-lg bg-[rgba(245,158,11,0.15)] text-[0.72rem] text-[#D97706] font-semibold border border-[rgba(245,158,11,0.3)] cursor-pointer"
   224→                  >
   225→                    <i className="fas fa-copy mr-1"></i>Copier
   226→                  </button>
   227→                </div>
   228→              </div>
   229→            )}
   230→
   231→            {/* 6-digit OTP input */}
   232→            <div className="flex justify-center gap-2 mb-5">
   233→              {[0, 1, 2, 3, 4, 5].map((i) => (
   234→                <input
   235→                  key={i}
   236→                  type="text"
   237→                  inputMode="numeric"
   238→                  maxLength={1}
   239→                  className="w-10 h-12 text-center text-[1.1rem] font-bold bg-white border-[1.5px] border-[rgba(0,0,0,0.1)] rounded-xl outline-none focus:border-[#22C55E] focus:shadow-[0_0_0_3px_rgba(34,197,94,0.1)] transition-all text-[#1F2937]"
   240→                  value={otpCode[i] || ''}
   241→                  onChange={(e) => {
   242→                    const val = e.target.value.replace(/\D/g, '');
   243→                    const newCode = otpCode.split('');
   244→                    newCode[i] = val.slice(-1);
   245→                    const final = newCode.join('').slice(0, 6);
   246→                    setOtpCode(final);
   247→                    if (val && i < 5) {
   248→                      const next = e.target.nextElementSibling as HTMLInputElement;
   249→                      next?.focus();
   250→                    }
   251→                  }}
   252→                  onKeyDown={(e) => {
   253→                    if (e.key === 'Backspace' && !otpCode[i] && i > 0) {
   254→                      const prev = (e.target as HTMLElement).previousElementSibling as HTMLInputElement;
   255→                      prev?.focus();
   256→                    }
   257→                  }}
   258→                  onPaste={(e) => {
   259→                    e.preventDefault();
   260→                    const paste = (e.clipboardData.getData('text') || '').replace(/\D/g, '').slice(0, 6);
   261→                    setOtpCode(paste);

   267→            <button

   272→              {otpLoading ? <div className="w-4 h-4 border-2 border-[rgba(255,255,255,0.3)] border-t-white rounded-full" style={{ animation: 'spin 0.6s linear infinite' }} /> : <><i className="fas fa-check-circle"></i> Vérifier</>}
   273→            </button>

   275→            <button
   276→              onClick={handleResendOtp}
   277→              disabled={otpLoading}
   278→              className="w-full py-3 rounded-xl border-[1.5px] border-[rgba(0,0,0,0.08)] bg-transparent text-[rgba(0,0,0,0.5)] font-semibold text-[0.82rem] cursor-pointer transition-transform active:scale-95 mb-3"
   279→            >
   280→              <i className="fas fa-redo mr-1.5 text-[0.7rem]"></i>Renvoyer le code
   281→            </button>
   282→
   283→            <button
   284→              onClick={() => { setOtpStep(false); setOtpCode(''); setSimCode(''); }}

   286→            >

   293→            <div className="flex bg-[#FFFFFF] rounded-xl p-[3px] mb-6 border border-[rgba(0,0,0,0.08)]">
   294→              <button onClick={() => { setMode('login'); setErrors({}); }} className={`flex-1 py-[11px] text-center text-[0.82rem] font-semibold rounded-lg transition-all border-none cursor-pointer ${mode === 'login' ? 'bg-[#22C55E] text-white shadow-lg' : 'text-[rgba(0,0,0,0.35)]'}`}>Connexion</button>
   295→              <button onClick={() => { setMode('register'); setErrors({}); }} className={`flex-1 py-[11px] text-center text-[0.82rem] font-semibold rounded-lg transition-all border-none cursor-pointer ${mode === 'register' ? 'bg-[#22C55E] text-white shadow-lg' : 'text-[rgba(0,0,0,0.35)]'}`}>Inscription</button>
   296→            </div>
   297→            {mode === 'login' ? (
   298→              <form onSubmit={handleLogin}>
   299→                <div className="mb-4 w-full"><label className="block mb-1.5 text-[0.75rem] font-semibold text-[rgba(0,0,0,0.45)]">Email</label><input name="email" type="email" required placeholder="votre@email.com" className="w-full premium-input" /></div>
   300→                <div className="mb-4 w-full relative"><label className="block mb-1.5 text-[0.75rem] font-semibold text-[rgba(0,0,0,0.45)]">Mot de passe</label><input name="password" type={showPw.l ? 'text' : 'password'} required placeholder="••••••••" className="w-full premium-input pr-11" /><button type="button" onClick={() => setShowPw({ ...showPw, l: !showPw.l })} className="absolute right-3 top-[38px] bg-transparent border-none text-[rgba(0,0,0,0.35)] cursor-pointer p-0.5"><i className={`fas ${showPw.l ? 'fa-eye-slash' : 'fa-eye'}`}></i></button></div>
   301→                <button type="submit" disabled={loading} className="w-full py-3.5 rounded-xl btn-gradient-green text-[0.88rem] cursor-pointer transition-transform active:scale-[0.97] disabled:opacity-60 flex items-center justify-center gap-2">{loading ? <div className="w-4 h-4 border-2 border-[rgba(255,255,255,0.3)] border-t-white rounded-full" style={{ animation: 'spin 0.6s linear infinite' }} /> : <><i className="fas fa-arrow-right"></i> Se connecter</>}</button>
   302→                <div className="mt-3"><a href="/forgot-password" className="text-[0.72rem] text-[#22C55E] font-medium hover:underline">Mot de passe oublié ?</a></div>

   304→            ) : (

   309→                <div className="mb-2.5 w-full relative"><label className="block mb-1 text-[0.72rem] font-semibold text-[rgba(0,0,0,0.45)]">Confirmer</label><input name="password2" type={showPw.r2 ?
