   311→// ===== Video player modal with NO seeking/scrolling =====
   312→function VideoPlayerModal({ video, onClose, onReward }: {
   313→  video: VideoItem;
   314→  onClose: () => void;
   315→  onReward: (reward: number) => Promise<void>;
   316→}) {
   317→  const playerRef = useRef<any>(null);
   318→  const containerRef = useRef<HTMLDivElement>(null);
   319→  const [watchedPercent, setWatchedPercent] = useState(0);
   320→  const [canClaim, setCanClaim] = useState(false);
   321→  const [claiming, setClaiming] = useState(false);
   322→  const [error, setError] = useState<string | null>(null);
   323→  const lastTimeRef = useRef(0);
   324→
   325→  // Load YouTube IFrame API
   326→  useEffect(() => {
   327→    if (!(window as any).YT) {
   328→      const tag = document.createElement('script');
   329→      tag.src = 'https://www.youtube.com/iframe_api';
   330→      document.body.appendChild(tag);
   331→    }
   332→
   333→    const initPlayer = () => {
   334→      const YT = (window as any).YT;
   335→      if (!YT || !YT.Player) {
   336→        setTimeout(initPlayer, 300);
   337→        return;
   338→      }
   339→      playerRef.current = new YT.Player('yt-player', {
   340→        videoId: video.id,
   341→        playerVars: {
   342→          autoplay: 1,
   343→          controls: 0,        // NO controls
   344→          disablekb: 1,       // NO keyboard
   345→          modestbranding: 1,
   346→          rel: 0,
   347→          playsinline: 1,
   348→          fs: 0,              // NO fullscreen
   349→          iv_load_policy: 3,  // NO annotations
   350→          nocookie: 1,
   351→        },
   352→        events: {
   353→          onReady: (e: any) => e.target.playVideo(),
   354→          onStateChange: (e: any) => {
   355→            // If user pauses, resume automatically
   356→            if (e.data === 2) {
   357→              setTimeout(() => playerRef.current?.playVideo(), 100);
   358→            }
   359→          },
   360→        },
   361→      });
   362→    };
   363→
   364→    const interval = setInterval(() => {
   365→      if (playerRef.current && playerRef.current.getCurrentTime) {
   366→        try {
   367→          const current = playerRef.current.getCurrentTime();
   368→          const duration = playerRef.current.getDuration();
   369→
   370→          // Anti-seeking: if time jumped forward abnormally, reset
   371→          if (current > lastTimeRef.current + 2) {
   372→            playerRef.current.seekTo(lastTimeRef.current, true);
   373→            return;
   374→          }
   375→          lastTimeRef.current = current;
   376→
   377→          if (duration > 0) {
   378→            const pct = (current / duration) * 100;
   379→            setWatchedPercent(pct);
   380→            if (pct >= 50) setCanClaim(true);
   381→          }
   382→        } catch { /* ignore */ }
   383→      }
   384→    }, 1000);
   385→
   386→    // Wait for YT API
   387→    if ((window as any).YT && (window as any).YT.Player) {
   388→      initPlayer();
   389→    } else {
   390→      (window as any).onYouTubeIframeAPIReady = initPlayer;
   391→    }
   392→
   393→    return () => {
   394→      clearInterval(interval);
   395→      if (playerRef.current && playerRef.current.destroy) {
   396→        playerRef.current.destroy();
   397→      }
   398→    };
   399→  }, [video.id]);
   400→
   401→  // Prevent scroll on the container
   402→  const preventScroll = (e: React.WheelEvent | React.TouchMoveEvent) => {
   403→    e.preventDefault();
   404→  };
   405→
   406→  const handleClaim = async () => {
   407→    setClaiming(true);
   408→    setError(null);
   409→    try {
   410→      const res = await authFetch('/api/videos/reward', {
   411→        method: 'POST',
   412→        headers: { 'Content-Type': 'application/json' },
   413→        body: JSON.stringify({ videoId: video.id, watchedPercent }),
   414→      });
   415→      const data = await res.json();
   416→      if (data.success) {
   417→        await onReward(video.reward);
   418→      } else {
   419→        setError(data.error || 'Erreur');
   420→      }
   421→    } catch {
   422→      setError('Erreur de connexion');
   423→    }
   424→    setClaiming(false);
   425→  };
   426→
   427→  return (
   428→    <div className="fixed inset-0 z-[9000] flex items-center justify-center p-4" style={{ background: 'rgba(0,0,0,0.9)' }}>
   429→      <div className="w-full max-w-[420px]">
   430→        {/* Header */}
   431→        <div className="flex items-center justify-between mb-3">
   432→          <div className="flex items-center gap-2">
   433→            <span className="text-[0.55rem] font-bold px-1.5 py-0.5 rounded text-white" style={{ background: CATEGORY_COLOR[video.category] }}>
   434→              {CATEGORY_FLAG[video.category]} {CATEGORY_LABEL[video.category]}
   435→            </span>
   436→            <span className="text-[0.6rem] text-white/60">Sponsorisé par {video.sponsor}</span>
   437→          </div>
   438→          {canClaim && (
   439→            <button onClick={onClose} className="text-white/60 hover:text-white cursor-pointer">
   440→              <i className="fas fa-times text-[1rem]"></i>
   441→            </button>
   442→          )}
   443→        </div>
   444→
   445→        {/* Title */}
   446→        <h3 className="text-[0.9rem] font-bold text-white mb-2">{video.title}</h3>
   447→
   448→        {/* Player container - NO scroll/seek allowed */}
   449→        <div
   450→          ref={containerRef}
   451→          onWheel={preventScroll}
   452→          onTouchMove={preventScroll}
   453→          className="relative w-full rounded-xl overflow-hidden bg-black"
   454→          style={{ aspectRatio: '16/9' }}
   455→        >
   456→          <div id="yt-player" className="w-full h-full" style={{ pointerEvents: 'none' }}></div>
   457→
   458→          {/* Overlay to block all interaction */}
   459→          <div className="absolute inset-0" style={{ pointerEvents: 'none' }}></div>
   460→
   461→          {/* Progress bar */}
   462→          <div className="absolute bottom-0 left-0 right-0 h-1.5 bg-white/20">
   463→            <div className="h-full transition-all" style={{ width: `${watchedPercent}%`, background: canClaim ? '#22C55E' : '#F59E0B' }}></div>
   464→          </div>
   465→
   466→          {/* Warning if trying to scroll */}
   467→          {watchedPercent < 50 && (
   468→            <div className="absolute top-2 left-2 right-2 text-center">
   469→              <div className="inline-block px-2 py-1 rounded-full text-[0.55rem] font-bold text-white" style={{ background: 'rgba(0,0,0,0.7)' }}>
