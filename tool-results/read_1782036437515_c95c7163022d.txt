   312→   312→function VideoPlayerModal({ video, onClose, onReward }: {
   313→   313→  video: VideoItem;
   314→   314→  onClose: () => void;
   315→   315→  onReward: (reward: number) => Promise<void>;
   316→   316→}) {
   317→   317→  const playerRef = useRef<any>(null);
   318→   318→  const containerRef = useRef<HTMLDivElement>(null);
   319→   319→  const [watchedPercent, setWatchedPercent] = useState(0);
   320→   320→  const [canClaim, setCanClaim] = useState(false);
   321→   321→  const [claiming, setClaiming] = useState(false);
   322→   322→  const [error, setError] = useState<string | null>(null);
   323→   323→  const lastTimeRef = useRef(0);
   324→   324→
   325→   325→  // Load YouTube IFrame API
   326→   326→  useEffect(() => {
   327→   327→    if (!(window as any).YT) {
   328→   328→      const tag = document.createElement('script');
   329→   329→      tag.src = 'https://www.youtube.com/iframe_api';
   330→   330→      document.body.appendChild(tag);
   331→   331→    }
   332→   332→
   333→   333→    const initPlayer = () => {
   334→   334→      const YT = (window as any).YT;
   335→   335→      if (!YT || !YT.Player) {
   336→   336→        setTimeout(initPlayer, 300);
   337→   337→        return;
   338→   338→      }
   339→   339→      playerRef.current = new YT.Player('yt-player', {
   340→   340→        videoId: video.id,
   341→   341→        playerVars: {
   342→   342→          autoplay: 1,
   343→   343→          controls: 0,        // NO controls
   344→   344→          disablekb: 1,       // NO keyboard
   345→   345→          modestbranding: 1,
   346→   346→          rel: 0,
   347→   347→          playsinline: 1,
   348→   348→          fs: 0,              // NO fullscreen
   349→   349→          iv_load_policy: 3,  // NO annotations
   350→   350→          nocookie: 1,
   351→   351→        },
   352→   352→        events: {
   353→   353→          onReady: (e: any) => e.target.playVideo(),
   354→   354→          onStateChange: (e: any) => {
   355→   355→            // If user pauses, resume automatically
   356→   356→            if (e.data === 2) {
   357→   357→              setTimeout(() => playerRef.current?.playVideo(), 100);
   358→   358→            }
   359→   359→          },
   360→   360→        },
   361→   361→      });
   362→   362→    };
   363→   363→
   364→   364→    const interval = setInterval(() => {
   365→   365→      if (playerRef.current && playerRef.current.getCurrentTime) {
   366→   366→        try {
   367→   367→          const current = playerRef.current.getCurrentTime();
   368→   368→          const duration = playerRef.current.getDuration();
   369→   369→
   370→   370→          // Anti-seeking: if time jumped forward abnormally, reset
   371→   371→          if (current > lastTimeRef.current + 2) {
   372→   372→            playerRef.current.seekTo(lastTimeRef.current, true);
   373→   373→            return;
   374→   374→          }
   375→   375→          lastTimeRef.current = current;
   376→   376→
   377→   377→          if (duration > 0) {
   378→   378→            const pct = (current / duration) * 100;
   379→   379→            setWatchedPercent(pct);
   380→   380→            if (pct >= 50) setCanClaim(true);
   381→   381→          }
   382→   382→        } catch { /* ignore */ }
   383→   383→      }
   384→   384→    }, 1000);
   385→   385→
   386→   386→    // Wait for YT API
   387→   387→    if ((window as any).YT && (window as any).YT.Player) {
   388→   388→      initPlayer();
   389→   389→    } else {
   390→   390→      (window as any).onYouTubeIframeAPIReady = initPlayer;
   391→   391→    }
   392→   392→
   393→   393→    return () => {
   394→   394→      clearInterval(interval);
   395→   395→      if (playerRef.current && playerRef.current.destroy) {
   396→   396→        playerRef.current.destroy();
