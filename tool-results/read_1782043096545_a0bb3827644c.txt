   313→// ===== Video player modal with NO seeking/scrolling =====
   314→function VideoPlayerModal({ video, onClose, onReward }: {
   315→  video: VideoItem;
   316→  onClose: () => void;
   317→  onReward: (reward: number) => Promise<void>;
   318→}) {
   319→  const playerRef = useRef<any>(null);
   320→  const containerRef = useRef<HTMLDivElement>(null);
   321→  const [watchedPercent, setWatchedPercent] = useState(0);
   322→  const [canClaim, setCanClaim] = useState(false);
   323→  const [claiming, setClaiming] = useState(false);
   324→  const [error, setError] = useState<string | null>(null);
   325→  const lastTimeRef = useRef(0);
   326→
   327→  // Load YouTube IFrame API
   328→  useEffect(() => {
   329→    if (!(window as any).YT) {
   330→      const tag = document.createElement('script');
   331→      tag.src = 'https://www.youtube.com/iframe_api';
   332→      document.body.appendChild(tag);
   333→    }
   334→
   335→    const initPlayer = () => {
   336→      const YT = (window as any).YT;
   337→      if (!YT || !YT.Player) {
   338→        setTimeout(initPlayer, 300);
   339→        return;
   340→      }
   341→      playerRef.current = new YT.Player('yt-player', {
   342→        videoId: video.id,
   343→        playerVars: {
   344→          autoplay: 1,
   345→          controls: 0,        // NO controls
   346→          disablekb: 1,       // NO keyboard
   347→          modestbranding: 1,
   348→          rel: 0,
   349→          playsinline: 1,
   350→          fs: 0,              // NO fullscreen
   351→          iv_load_policy: 3,  // NO annotations
   352→          nocookie: 1,
   353→        },
   354→        events: {
   355→          onReady: (e: any) => e.target.playVideo(),
   356→          onStateChange: (e: any) => {
   357→            // If user pauses, resume automatically
   358→            if (e.data === 2) {
   359→              setTimeout(() => playerRef.current?.playVideo(), 100);
   360→            }
   361→          },
   362→        },
   363→      });
   364→    };
   365→
   366→    const interval = setInterval(() => {
   367→      if (playerRef.current && playerRef.current.getCurrentTime) {
   368→        try {
   369→          const current = playerRef.current.getCurrentTime();
   370→          const duration = playerRef.current.getDuration();
   371→
   372→          // Anti-seeking: if time jumped forward abnormally, reset
   373→          if (current > lastTimeRef.current + 2) {
   374→            playerRef.current.seekTo(lastTimeRef.current, true);
   375→            return;
   376→          }
   377→          lastTimeRef.current = current;
   378→
   379→          if (duration > 0) {
   380→            const pct = (current / duration) * 100;
   381→            setWatchedPercent(pct);
   382→            if (pct >= 50) setCanClaim(true);
   383→          }
   384→        } catch { /* ignore */ }
   385→      }
   386→    }, 1000);
   387→
   388→    // Wait for YT API
   389→    if ((window as any).YT && (window as any).YT.Player) {
   390→      initPlayer();
   391→    } else {
   392→      (window as any).onYouTubeIframeAPIReady = initPlayer;
   393→    }
   394→
   395→    return () => {
   396→      clearInterval(interval);
   397→      if (playerRef.current && playerRef.current.destroy) {
   398→        playerRef.current.destroy();
   399→      }
   400→    };
   401→  }, [video.id]);
   402→
   403→  // Prevent scroll on the container
   404→  const preventScroll = (e: React.WheelEvent | React.TouchEvent) => {
   405→    e.preventDefault();
   406→  };
   407→
   408→  const handleClaim = async () => {
   409→    setClaiming(true);
   410→    setError(null);
   411→    try {
   412→      const res = await authFetch('/api/videos/reward', {
   413→        method: 'POST',
   414→        headers: { 'Content-Type': 'application/json' },
   415→        body: JSON.stringify({ videoId: video.id, watchedPercent }),
   416→      });
   417→      const data = await res.json();
   418→      if (data.success) {
   419→        await onReward(video.reward);
   420→      } else {
   421→        setError(data.error || 'Erreur');
   422→      }
   423→    } catch {
   424→      setError('Erreur de connexion');
   425→    }
   426→    setClaiming(false);
   427→  };
   428→
   429→  return (
   430→    <div className="fixed inset-0 z-[9000] flex items-center justify-center p-4" style={{ background: 'rgba(0,0,0,0.9)' }}>
   431→      <div className="w-full max-w-[420px]">
   432→        {/* Header — Quit (X) button ALWAYS visible (top-right, 44px touch target) */}
   433→        <div className="flex items-center justify-between mb-3 gap-2">
   434→          <div className="flex items-center gap-2 min-w-0 flex-1">
   435→            <span className="text-[0.55rem] font-bold px-1.5 py-0.5 rounded text-white flex-shrink-0" style={{ background: CATEGORY_COLOR[video.category] }}>
   436→              {CATEGORY_FLAG[video.category]} {CATEGORY_LABEL[video.category]}
   437→            </span>
   438→            <span className="text-[0.6rem] text-white/60 truncate">Sponsorisé par {video.sponsor}</span>
   439→          </div>
   440→          {/* Always-visible Quit button — anyone can quit at any moment */}
   441→          <button
   442→            onClick={onClose}
   443→            aria-label="Quitter la vidéo"
   444→            title="Quitter la vidéo"
   445→            className="flex-shrink-0 w-11 h-11 rounded-xl flex items-center justify-center cursor-pointer transition-all active:scale-95 border-none"
   446→            style={{ background: 'rgba(239,68,68,0.95)', color: '#FFFFFF', boxShadow: '0 2px 10px rgba(239,68,68,0.45)' }}
   447→          >
   448→            <i className="fas fa-times text-[1.15rem]"></i>
   449→          </button>
   450→        </div>
   451→
   452→        {/* Title */}
   453→        <h3 className="text-[0.9rem] font-bold text-white mb-2">{video.title}</h3>
   454→
   455→        {/* Player container - NO scroll/seek allowed */}
   456→        <div
   457→          ref={containerRef}
   458→          onWheel={preventScroll}
   459→          onTouchMove={preventScroll}
   460→          className="relative w-full rounded-xl overflow-hidden bg-black"
   461→          style={{ aspectRatio: '16/9' }}
   462→        >
   463→          <div id="yt-player" className="w-full h-full" style={{ pointerEvents: 'none' }}></div>
   464→
   465→          {/* Transparent overlay that CAPTURES all pointer events so the user
   466→              cannot click the YouTube iframe to seek/pause/interact. */}
