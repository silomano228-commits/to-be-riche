     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useAppStore, formatMoney, authFetch, refreshUser as globalRefreshUser } from '@/lib/store';
     5→import { Header, INVEST_LEVELS } from '@/components/shared';
     6→import { CongratulationsModal, type CongratulationsData } from '@/components/CongratulationsModal';
     7→
     8→// ---------- helpers ----------
     9→const hexToRgba = (hex: string, alpha: number) => {
    10→  const r = parseInt(hex.slice(1, 3), 16);
    11→  const g = parseInt(hex.slice(3, 5), 16);
    12→  const b = parseInt(hex.slice(5, 7), 16);
    13→  return `rgba(${r},${g},${b},${alpha})`;
    14→};
    15→
    16→type PaymentMethod = 'yas' | 'trx';
    17→type PayoutMethod = 'main' | 'yas_trx';
    18→
    19→interface InvestSummary {
    20→  total: number;
    21→  active: number;
    22→  completed: number;
    23→  totalEarned: number;
    24→  totalInvested: number;
    25→}
    26→
    27→export default function InvestHubScreen() {
    28→  const { user, addToast } = useAppStore();
    29→  const [investments, setInvestments] = useState<any[]>([]);
    30→  const [summary, setSummary] = useState<InvestSummary | null>(null);
    31→  const [loading, setLoading] = useState(true);
    32→  const [now, setNow] = useState(Date.now());
    33→
    34→  // Create-investment modal state
    35→  const [showCreate, setShowCreate] = useState<number | null>(null);
    36→  const [createAmt, setCreateAmt] = useState('');
    37→  const [createPayment, setCreatePayment] = useState<PaymentMethod>('yas');
    38→  const [createAddress, setCreateAddress] = useState('');
    39→  const [creating, setCreating] = useState(false);
    40→
    41→  // Unlock modal state
    42→  const [unlockLevel, setUnlockLevel] = useState<number | null>(null);
    43→  const [unlocking, setUnlocking] = useState(false);
    44→
    45→  // Claim payout modal state
    46→  const [claimTarget, setClaimTarget] = useState<{ id: string; level: number; amount: number; rate: number; gain: number } | null>(null);
    47→  const [payoutChoice, setPayoutChoice] = useState<PayoutMethod>('main');
    48→  const [claimPaymentType, setClaimPaymentType] = useState<PaymentMethod>('yas');
    49→  const [claimAddress, setClaimAddress] = useState('');
    50→  const [claiming, setClaiming] = useState(false);
    51→
    52→  // Congratulations modal
    53→  const [congrats, setCongrats] = useState<CongratulationsData>({ show: false, type: 'collect' });
    54→
    55→  // ---------- data loading ----------
    56→  useEffect(() => {
    57→    const t = setInterval(() => setNow(Date.now()), 1000);
    58→    return () => clearInterval(t);
    59→  }, []);
    60→
    61→  const loadInvestments = useCallback(async () => {
    62→    try {
    63→      const res = await authFetch('/api/invest/list');
    64→      const data = await res.json();
    65→      if (data.success) {
    66→        setInvestments(data.investments || []);
    67→        setSummary(data.summary || null);
    68→      }
    69→    } catch { /* */ }
    70→    setLoading(false);
    71→  }, []);
    72→
    73→  useEffect(() => {
    74→    const t = setTimeout(() => { loadInvestments(); }, 0);
    75→    return () => clearTimeout(t);
    76→  }, [loadInvestments]);
    77→
    78→  const refreshUser = async () => { await globalRefreshUser(); };
    79→
    80→  // ---------- create investment ----------
    81→  const openCreate = (level: number) => {
    82→    setShowCreate(level);
    83→    setCreateAmt('');
    84→    setCreatePayment('yas');
    85→    setCreateAddress('');
    86→  };
    87→
    88→  const handleCreate = async () => {
    89→    if (showCreate == null) return;
    90→    const level = showCreate;
    91→    const lvl = INVEST_LEVELS[level - 1];
    92→    if (!lvl) return;
    93→    const amt = parseFloat(createAmt);
    94→    if (!amt || amt < lvl.min || amt > lvl.max) {
    95→      addToast(`Montant: $${lvl.min} - $${lvl.max}`, 'error');
    96→      return;
    97→    }
    98→    if (!createAddress.trim()) {
    99→      addToast('Adresse requise', 'error');
   100→      return;
   101→    }
   102→    setCreating(true);
   103→    try {
   104→      const res = await authFetch('/api/invest/create', {
   105→        method: 'POST',
   106→        headers: { 'Content-Type': 'application/json' },
   107→        body: JSON.stringify({
   108→          level,
   109→          amount: amt,
   110→          paymentMethod: createPayment,
   111→          userAddress: createAddress.trim(),
   112→        }),
   113→      });
   114→      const data = await res.json();
   115→      if (data.success) {
   116→        setShowCreate(null);
   117→        loadInvestments();
   118→        refreshUser();
   119→        setCongrats({
   120→          show: true,
   121→          type: 'generic',
   122→          title: 'Investissement créé !',
   123→          amount: amt,
   124→          message: data.message || `Votre investissement Niveau ${level} (${lvl.name}) de ${formatMoney(amt)} est actif. Paiement ${createPayment.toUpperCase()} en cours — fonds disponibles dans les 6 heures. Collecte quotidienne illimitée !`,
   125→          onClose: () => setCongrats(c => ({ ...c, show: false })),
   126→        });
   127→      } else {
   128→        addToast(data.error || 'Erreur', 'error');
   129→      }
   130→    } catch {
   131→      addToast('Erreur réseau', 'error');
   132→    }
   133→    setCreating(false);
   134→  };
   135→
   136→  // ---------- unlock level ----------
   137→  const handleUnlock = async (level: number) => {
   138→    if (unlocking) return;
   139→    setUnlocking(true);
   140→    try {
   141→      const res = await authFetch('/api/invest/unlock', {
   142→        method: 'POST',
   143→        headers: { 'Content-Type': 'application/json' },
   144→        body: JSON.stringify({ level }),
   145→      });
   146→      const data = await res.json();
   147→      if (data.success) {
   148→        addToast(data.message || 'Niveau débloqué !', 'success');
   149→        setUnlockLevel(null);
   150→        refreshUser();
   151→      } else {
   152→        addToast(data.error || 'Erreur', 'error');
   153→      }
   154→    } catch {
   155→      addToast('Erreur réseau', 'error');
   156→    }
   157→    setUnlocking(false);
   158→  };
   159→
   160→  // ---------- claim payout ----------
   161→  const openClaim = (inv: any) => {
   162→    const lvl = INVEST_LEVELS[inv.level - 1];
   163→    const gain = Math.round(inv.amount * inv.rate / 100 * 100) / 100;
   164→    setClaimTarget({ id: inv.id, level: inv.level, amount: inv.amount, rate: inv.rate, gain });
   165→    setPayoutChoice(gain >= 5 ? 'main' : 'main'); // default to main; YAS/TRX only enabled if gain >= 5
   166→    setClaimPaymentType('yas');
   167→    setClaimAddress('');
   168→  };
   169→
   170→  const handleClaim = async () => {
   171→    if (!claimTarget) return;
   172→    const method: PayoutMethod = payoutChoice;
   173→    if (method === 'yas_trx') {
   174→      if (claimTarget.gain < 5) {
   175→        addToast('Gain insuffisant pour YAS/TRX (min $5)', 'error');
   176→        return;
   177→      }
   178→      if (!claimAddress.trim()) {
   179→        addToast('Adresse de retrait requise', 'error');
   180→        return;
   181→      }
   182→    }
   183→    setClaiming(true);
   184→    try {
   185→      const res = await authFetch('/api/invest/claim', {
   186→        method: 'POST',
   187→        headers: { 'Content-Type': 'application/json' },
   188→        body: JSON.stringify({
   189→          investmentId: claimTarget.id,
   190→          payoutMethod: method,
   191→          userAddress: method === 'yas_trx' ? claimAddress.trim() : undefined,
   192→          paymentType: method === 'yas_trx' ? claimPaymentType : undefined,
   193→        }),
   194→      });
   195→      const data = await res.json();
   196→      if (data.success) {
   197→        const target = claimTarget;
   198→        setClaimTarget(null);
   199→        loadInvestments();
   200→        refreshUser();
   201→        setCongrats({
   202→          show: true,
   203→          type: 'collect',
   204→          amount: data.gain,
   205→          title: 'Collecte réussie !',
   206→          message: data.message || `Vous avez réclamé ${formatMoney(data.gain)} ${method === 'main' ? 'versé sur votre compte principal.' : '— retrait demandé, fonds dans les 6h.'}`,
   207→          onClose: () => setCongrats(c => ({ ...c, show: false })),
   208→        });
   209→        void target;
   210→      } else {
   211→        addToast(data.error || 'Erreur', 'error');
   212→      }
   213→    } catch {
   214→      addToast('Erreur réseau', 'error');
   215→    }
   216→    setClaiming(false);
   217→  };
   218→
   219→  if (!user) return null;
   220→
   221→  const unlockedLevel = (user as any).unlockedLevel || 1;
   222→  const referralCount = (user as any).referralCount || 0;
   223→  const activeInv = investments.filter(i => i.status === 'active');
   224→  const completedInv = investments.filter(i => i.status === 'completed');
   225→  const investedLevels = new Set(investments.map(i => i.level));
   226→
   227→  // Has the user invested in the previous level? (required to unlock/invest higher)
   228→  const hasPrevLevel = (level: number) => level === 1 || investedLevels.has(level - 1);
   229→
   230→  // Referral info per locked level
   231→  const getUnlockInfo = (level: number) => {
   232→    const lvl = INVEST_LEVELS[level - 1];
   233→    if (!lvl) return null;
   234→    const required = lvl.requiredReferrals;
   235→    const missing = Math.max(0, required - referralCount);
   236→    return { required, missing, canUnlock: missing === 0 };
   237→  };
   238→
   239→  // ---------- render ----------
   240→  return (
   241→    <>
   242→      <style>{`
   243→        @keyframes slideUp { from { opacity: 0; transform: translateY(8px); } to { opacity: 1; transform: translateY(0); } }
   244→        @keyframes modalIn { from { opacity: 0; transform: scale(0.95); } to { opacity: 1; transform: scale(1); } }
   245→        @keyframes claimPulse { 0%,100% { box-shadow: 0 0 0 0 rgba(34,197,94,0.45); transform: scale(1); } 50% { box-shadow: 0 0 0 10px rgba(34,197,94,0); transform: scale(1.03); } }
   246→        @keyframes timerColon { 0%,100% { opacity: 1; } 50% { opacity: 0.3; } }
   247→        @keyframes heroGlow { 0%,100% { box-shadow: 0 0 24px rgba(34,197,94,0.12); } 50% { box-shadow: 0 0 40px rgba(20,184,166,0.18); } }
   248→        .claim-pulse { animation: claimPulse 1.6s ease-in-out infinite; }
   249→        .timer-colon { animation: timerColon 1s step-end infinite; }
   250→        .hero-glow { animation: heroGlow 3.5s ease-in-out infinite; }
   251→        .pm-card { transition: transform .15s ease, box-shadow .15s ease; }
   252→        .pm-card:active { transform: scale(0.98); }
   253→      `}</style>
   254→
   255→      <Header
   256→        title="Investir"
   257→        icon="fa-chart-line"
   258→        iconColor="#22C55E"
   259→        leftElement={
   260→          <button
   261→            onClick={() => useAppStore.getState().setPage('home')}
   262→            className="w-9 h-9 rounded-full flex items-center justify-center cursor-pointer border-none mr-1"
   263→            style={{ background: 'rgba(0,0,0,0.06)', color: 'rgba(0,0,0,0.55)' }}
   264→            aria-label="Retour"
   265→          >
   266→            <i className="fas fa-arrow-left text-[0.8rem]"></i>
   267→          </button>
   268→        }
   269→      />
   270→
   271→      <div className="px-[18px] py-4 flex-1 w-full overflow-y-auto min-h-0" style={{ background: '#F6F8F7' }}>
   272→
   273→        {/* Hero summary card (replaces investBalance) */}
   274→        <div
   275→          className="hero-glow rounded-2xl p-5 mb-4 relative overflow-hidden"
   276→          style={{
   277→            background: 'linear-gradient(135deg, #16A34A 0%, #14B8A6 100%)',
   278→            color: '#FFFFFF',
   279→          }}
   280→        >
   281→          <div className="absolute -top-10 -right-10 w-32 h-32 rounded-full" style={{ background: 'rgba(255,255,255,0.08)' }}></div>
   282→          <div className="absolute -bottom-8 -left-8 w-24 h-24 rounded-full" style={{ background: 'rgba(255,255,255,0.06)' }}></div>
   283→          <div className="absolute top-3 right-4 text-[0.6rem] font-mono opacity-50">INVEST·HUB</div>
   284→
   285→          <div className="flex items-center gap-2 mb-1">
   286→            <div className="w-8 h-8 rounded-lg flex items-center justify-center" style={{ background: 'rgba(255,255,255,0.18)' }}>
   287→              <i className="fas fa-seedling text-[0.85rem] text-white"></i>
   288→            </div>
   289→            <div className="text-[0.7rem] font-semibold uppercase tracking-[1.5px] opacity-90">Vos gains investis</div>
   290→          </div>
   291→
   292→          <div className="text-[2.1rem] font-black mb-3 tracking-tight leading-none">
   293→            +{formatMoney(summary?.totalEarned ?? 0)}
   294→          </div>
   295→
   296→          <div className="grid grid-cols-2 gap-3 text-[0.7rem]">
   297→            <div className="rounded-xl p-2.5" style={{ background: 'rgba(255,255,255,0.12)' }}>
   298→              <div className="opacity-80 uppercase tracking-wide text-[0.6rem]">Investi</div>
   299→              <div className="font-bold text-[0.95rem]">{formatMoney(summary?.totalInvested ?? 0)}</div>
   300→            </div>
   301→            <div className="rounded-xl p-2.5" style={{ background: 'rgba(255,255,255,0.12)' }}>
   302→              <div className="opacity-80 uppercase tracking-wide text-[0.6rem]">Actifs</div>
   303→              <div className="font-bold text-[0.95rem]">{summary?.active ?? 0}</div>
   304→            </div>
   305→          </div>
   306→        </div>
   307→
   308→        {/* Info banner: unlimited collection */}
   309→        <div
   310→          className="rounded-xl p-3 mb-4 flex items-start gap-2.5"
   311→          style={{ background: 'rgba(20,184,166,0.08)', border: '1px solid rgba(20,184,166,0.18)' }}
   312→        >
   313→          <i className="fas fa-infinity text-[0.85rem] mt-0.5" style={{ color: '#14B8A6' }}></i>
   314→          <div>
   315→            <div className="text-[0.78rem] font-bold" style={{ color: '#0F766E' }}>Collecte quotidienne illimitée</div>
   316→            <div className="text-[0.7rem] mt-0.5" style={{ color: 'rgba(15,118,110,0.85)' }}>
   317→              Tous les niveaux rapportent 10%/jour. Le dépôt se fait directement par YAS ou TRX.
   318→            </div>
   319→          </div>
   320→        </div>
   321→
   322→        {/* Claimable alert */}
   323→        {activeInv.some(i => i.canClaim) && (
   324→          <div
   325→            className="rounded-xl p-3.5 mb-4 flex items-center gap-3"
   326→            style={{ background: 'rgba(34,197,94,0.1)', border: '1px solid rgba(34,197,94,0.18)', animation: 'slideUp 0.3s ease-out' }}
   327→          >
   328→            <div className="w-10 h-10 rounded-xl flex items-center justify-center shrink-0" style={{ background: 'rgba(34,197,94,0.15)' }}>
   329→              <i className="fas fa-gift text-[0.95rem]" style={{ color: '#22C55E' }}></i>
   330→            </div>
   331→            <div className="flex-1">
   332→              <div className="text-[0.82rem] font-bold" style={{ color: '#15803D' }}>Gains à réclamer !</div>
   333→              <div className="text-[0.7rem]" style={{ color: 'rgba(0,0,0,0.55)' }}>Faites défiler jusqu'à vos investissements actifs.</div>
   334→            </div>
   335→          </div>
   336→        )}
   337→
   338→        {/* ---------- Investment levels ---------- */}
   339→        <h3 className="text-[0.88rem] font-bold mb-2.5 flex items-center gap-1.5" style={{ color: '#1F2937' }}>
   340→          <i className="fas fa-layer-group text-[0.75rem]" style={{ color: '#14B8A6' }}></i>
   341→          Niveaux d'investissement
   342→        </h3>
   343→        <div className="space-y-3 mb-5">
   344→          {INVEST_LEVELS.map((lvl) => {
   345→            const isUnlocked = lvl.level <= unlockedLevel;
   346→            const prevOk = hasPrevLevel(lvl.level);
   347→            const canInvest = isUnlocked && prevOk;
   348→            const info = getUnlockInfo(lvl.level);
   349→
   350→            return (
   351→              <div
   352→                key={lvl.level}
   353→                className="rounded-2xl overflow-hidden relative"
   354→                style={{
   355→                  background: canInvest ? '#FFFFFF' : '#F3F4F6',
   356→                  border: `1px solid ${canInvest ? hexToRgba(lvl.color, 0.22) : 'rgba(0,0,0,0.06)'}`,
   357→                  animation: 'slideUp 0.3s ease-out',
   358→                }}
   359→              >
   360→                <div className="p-4">
   361→                  {/* Header row */}
   362→                  <div className="flex items-center gap-3 mb-3">
   363→                    <div
   364→                      className="w-11 h-11 rounded-xl flex items-center justify-center shrink-0"
   365→                      style={{ background: canInvest ? hexToRgba(lvl.color, 0.14) : 'rgba(0,0,0,0.06)' }}
   366→                    >
   367→                      <i
   368→                        className={`fas ${isUnlocked ? lvl.icon : 'fa-lock'} text-[1rem]`}
   369→                        style={{ color: canInvest ? lvl.color : 'rgba(0,0,0,0.3)' }}
   370→                      ></i>
   371→                    </div>
   372→                    <div className="flex-1 min-w-0">
   373→                      <div className="text-[0.92rem] font-bold truncate" style={{ color: canInvest ? '#1F2937' : 'rgba(0,0,0,0.5)' }}>
   374→                        Niv. {lvl.level} — {lvl.name}
   375→                      </div>
   376→                      <div className="text-[0.68rem] mt-0.5 flex items-center gap-1.5 flex-wrap" style={{ color: 'rgba(0,0,0,0.55)' }}>
   377→                        <span><i className="fas fa-coins text-[0.55rem] mr-0.5"></i>${lvl.min}-${lvl.max}</span>
   378→                        <span style={{ color: '#16A34A' }}><i className="fas fa-arrow-trend-up text-[0.55rem] mr-0.5"></i>{lvl.rate}%/jour</span>
   379→                      </div>
   380→                    </div>
   381→                  </div>
   382→
   383→                  {/* Badges row */}
   384→                  <div className="flex items-center gap-1.5 flex-wrap mb-3">
   385→                    <span
   386→                      className="text-[0.62rem] font-semibold px-2 py-1 rounded-md flex items-center gap-1"
   387→                      style={{ background: 'rgba(20,184,166,0.1)', color: '#0F766E' }}
   388→                    >
   389→                      <i className="fas fa-infinity text-[0.5rem]"></i>Collecte illimitée
   390→                    </span>
   391→                    {lvl.requiredReferrals > 0 && (
   392→                      <span
   393→                        className="text-[0.62rem] font-semibold px-2 py-1 rounded-md flex items-center gap-1"
   394→                        style={{
   395→                          background: isUnlocked ? 'rgba(34,197,94,0.1)' : 'rgba(0,0,0,0.05)',
   396→                          color: isUnlocked ? '#16A34A' : 'rgba(0,0,0,0.5)',
   397→                        }}
   398→                      >
   399→                        <i className="fas fa-users text-[0.5rem]"></i>{lvl.requiredReferrals} parrainés
   400→                        {isUnlocked && <i className="fas fa-check text-[0.5rem] ml-0.5"></i>}
   401→                      </span>
   402→                    )}
   403→                    {lvl.level === 1 && (
   404→                      <span className="text-[0.62rem] font-semibold px-2 py-1 rounded-md flex items-center gap-1" style={{ background: 'rgba(34,197,94,0.1)', color: '#16A34A' }}>
   405→                        <i className="fas fa-unlock text-[0.5rem]"></i>Libre
   406→                      </span>
   407→                    )}
   408→                  </div>
   409→
   410→                  {/* Action button */}
   411→                  {canInvest ? (
   412→                    <button
   413→                      onClick={() => openCreate(lvl.level)}
   414→                      className="w-full py-2.5 rounded-xl text-[0.82rem] font-bold border-none cursor-pointer transition-all active:scale-[0.98] flex items-center justify-center gap-2"
   415→                      style={{ background: lvl.color, color: '#FFFFFF', boxShadow: `0 3px 12px ${hexToRgba(lvl.color, 0.28)}` }}
   416→                    >
   417→                      <i className="fas fa-hand-holding-dollar text-[0.7rem]"></i>Investir
   418→                    </button>
   419→                  ) : !isUnlocked ? (
   420→                    <button
   421→                      onClick={() => setUnlockLevel(lvl.level)}
   422→                      className="w-full py-2.5 rounded-xl text-[0.78rem] font-semibold border-none cursor-pointer transition-all active:scale-[0.98] flex items-center justify-center gap-2"
   423→                      style={{ background: hexToRgba(lvl.color, 0.1), color: lvl.color, border: `1px solid ${hexToRgba(lvl.color, 0.22)}` }}
   424→                    >
   425→                      <i className="fas fa-lock-open text-[0.7rem]"></i>Débloquer
   426→                      {info && info.missing > 0 && (
   427→                        <span className="opacity-80 text-[0.7rem]">· {referralCount}/{info.required}</span>
   428→                      )}
   429→                    </button>
   430→                  ) : (
   431→                    <div
   432→                      className="w-full py-2.5 rounded-xl text-[0.72rem] font-semibold text-center flex items-center justify-center gap-1.5"
   433→                      style={{ background: 'rgba(0,0,0,0.04)', color: 'rgba(0,0,0,0.45)' }}
   434→                    >
   435→                      <i className="fas fa-arrow-up text-[0.6rem]"></i>Investissez d'abord au Niv. {lvl.level - 1}
   436→                    </div>
   437→                  )}
   438→                </div>
   439→              </div>
   440→            );
   441→          })}
   442→        </div>
   443→
   444→        {/* ---------- Active investments (vertical cards) ---------- */}
   445→        {activeInv.length > 0 && (
   446→          <>
   447→            <h3 className="text-[0.88rem] font-bold mb-2.5 flex items-center gap-1.5" style={{ color: '#1F2937' }}>
   448→              <i className="fas fa-bolt text-[0.75rem]" style={{ color: '#F59E0B' }}></i>
   449→              Investissements actifs
   450→            </h3>
   451→            <div className="space-y-3 mb-5">
   452→              {activeInv.map((inv) => {
   453→                const lvl = INVEST_LEVELS[inv.level - 1];
   454→                const nextMs = inv.nextClaimAt ? new Date(inv.nextClaimAt).getTime() - now : (inv.nextClaimInMs ?? 0);
   455→                const canClaim = inv.canClaim || nextMs <= 0;
