   158→function Pill({ icon, text }: { icon: string; text: string }) {
   159→  return (
   160→    <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full bg-white/20 backdrop-blur-sm text-white text-[0.62rem] font-semibold">

   166→
   167→/** A small subsection heading inside an accordion body. */
   168→function SubHead({ title, color }: { title: string; color: string }) {
   169→  return (
   170→    <div className="flex items-center gap-2 mb-1 mt-3 first:mt-0">
   171→      <div className="h-3 w-1 rounded-full" style={{ background: color }} />
   172→      <h4 className="text-[0.7rem] font-black text-[#1F2937] uppercase tracking-wide">{title}</h4>
   173→    </div>
   174→  );
   175→}
   176→
   177→/** A bullet row with a colored icon. */
   178→function Row({ icon, color, title, children }: { icon: string; color: string; title?: string; children: React.ReactNode }) {
   179→  return (
   180→    <div className="flex items-start gap-2.5 py-2">
   181→      <div className="w-7 h-7 rounded-lg flex items-center justify-center shrink-0 mt-0.5" style={{ background: `${color}1A` }}>
   182→        <i className={`fas ${icon} text-[0.7rem]`} style={{ color }} />
   183→      </div>
   184→      <div className="flex-1 text-[0.73rem] text-[#374151] leading-relaxed">
   185→        {title && <span className="font-bold text-[#1F2937]">{title} · </span>}
   186→        {children}
   187→      </div>
   188→    </div>
   189→  );
   190→}
   191→
   192→/** A highlighted callout box. */
   193→function Callout({ icon, color, children }: { icon: string; color: string; children: React.ReactNode }) {
   194→  return (
   195→    <div className="mt-3 rounded-xl p-3 flex items-start gap-2.5" style={{ background: `${color}12`, border: `1px solid ${color}33` }}>
   196→      <i className={`fas ${icon} text-[0.8rem] mt-0.5 shrink-0`} style={{ color }} />
   197→      <div className="text-[0.7rem] text-[#374151] leading-relaxed">{children}</div>
   198→    </div>
   199→  );
   200→}
   201→
   202→/** A simple key/value badge row (used for payment methods, levels, etc.). */
   203→function StatRow({ label, value, valueColor }: { label: string; value: string; valueColor?: string }) {
   204→  return (
   205→    <div className="flex items-center justify-between py-1.5 text-[0.72rem]">
   206→      <span className="text-[#6B7280]">{label}</span>
   207→      <span className="font-bold" style={{ color: valueColor || '#1F2937' }}>{value}</span>
   208→    </div>
   209→  );
   210→}
