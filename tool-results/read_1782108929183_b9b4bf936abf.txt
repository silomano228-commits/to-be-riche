    11→// Yas number validation
    12→function validateYasAccount(account: string): string | null {
    13→  const trimmed = account.trim();
    14→  if (!trimmed) return 'Numéro de compte Yas requis';
    15→  if (!/^\d{8}$/.test(trimmed)) return 'Le numéro doit contenir exactement 8 chiffres';
    16→  const prefix = trimmed.substring(0, 2);
    17→  if (!['90', '91', '92', '93', '70', '71', '72', '73'].includes(prefix)) {
    18→    return 'Le numéro doit commencer par 90-93 ou 70-73';
    19→  }
    20→  return null;
    21→}
