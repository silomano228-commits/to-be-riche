   481→  const handleEditBalance = async (targetUserId: string) => {
   482→    if (editBalanceSending || !editBalanceDraft) return;
   483→    const fieldLabels: Record<string, string> = {
   484→      balance: 'Solde principal',
   485→      videoBalance: 'Vidéo',
   486→      tradeBalance: 'Trading',
   487→      projectBalance: 'Projet',
   488→    };
   489→    // Detect changed fields and validate every input (must parse to >=0 numbers).
   490→    type FieldKey = keyof BalanceDraft;
   491→    const changes: { field: FieldKey; amount: number }[] = [];
   492→    for (const k of Object.keys(editBalanceDraft) as FieldKey[]) {
   493→      const raw = editBalanceDraft[k];
   494→      const parsed = parseFloat(raw);
   495→      if (isNaN(parsed) || parsed < 0) {
   496→        addToast(`${fieldLabels[k]} : montant invalide`, 'error');
   497→        return;
   498→      }
   499→      // Compare against the user's current value (from adminData). We treat
   500→      // any numeric difference as a change so trailing zeros don't cause
   501→      // spurious no-op API calls.
   502→      const current = Number((adminData?.users || []).find((u: any) => u.id === targetUserId)?.[k] ?? 0);
   503→      if (Math.abs(parsed - current) > 0.0001) {
   504→        changes.push({ field: k, amount: parsed });
   505→      }
   506→    }
   507→    if (changes.length === 0) {
   508→      addToast('Aucune modification', 'info');
   509→      setEditBalanceUserId(null);
   510→      setEditBalanceDraft(null);
   511→      return;
   512→    }
   513→    setEditBalanceSending(true);
   514→    let failures = 0;
   515→    for (const c of changes) {
   516→      try {
   517→        const res = await authFetch('/api/admin/update-balance', {
   518→          method: 'POST',
   519→          headers: { 'Content-Type': 'application/json' },
   520→          body: JSON.stringify({ userId: targetUserId, field: c.field, amount: c.amount }),
   521→        });
   522→        const data = await res.json();
   523→        if (!data.success) {
   524→          failures++;
   525→          addToast(`${fieldLabels[c.field]} : ${data.error || 'Erreur'}`, 'error');
   526→        }
   527→      } catch {
   528→        failures++;
   529→        addToast(`${fieldLabels[c.field]} : erreur réseau`, 'error');
