    14→  return `rgba(${r},${g},${b},${alpha})`;
    15→};

   290→          <div className="absolute -bottom-8 -left-8 w-24 h-24 rounded-full" style={{ background: 'rgba(255,255,255,0.06)' }}></div>
   291→          <div className="absolute top-3 right-4 text-[0.6rem] font-mono opacity-50">INVEST·HUB</div>

   293→          <div className="flex items-center gap-2 mb-1">

   295→              <i className="fas fa-seedling text-[0.85rem] text-white"></i>
