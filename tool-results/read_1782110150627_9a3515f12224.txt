   159→function Pill({ icon, text }: { icon: string; text: string }) {
   160→  return (
   161→    <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full bg-white/20 backdrop-blur-sm text-white text-[0.62rem] font-semibold">

   167→
   168→/** A small subsection heading inside an accordion body. */
   169→function SubHead({ title, color }: { title: string; color: string }) {
   170→  return (
   171→    <div className="flex items-center gap-2 mb-1 mt-3 first:mt-0">
   172→      <div className="h-3 w-1 rounded-full" style={{ background: color }} />
   173→      <h4 className="text-[0.7rem] font-black text-[#1F2937] uppercase tracking-wide">{title}</h4>
   174→    </div>
   175→  );
   176→}
   177→
   178→/** A bullet row with a colored icon. */
   179→function Row({ icon, color, title, children }: { icon: string; color: string; title?: string; children: React.ReactNode }) {
   180→  return (
   181→    <div className="flex items-start gap-2.5 py-2">
   182→      <div className="w-7 h-7 rounded-lg flex items-center justify-center shrink-0 mt-0.5" style={{ background: `${color}1A` }}>
   183→        <i className={`fas ${icon} text-[0.7rem]`} style={{ color }} />
   184→      </div>
   185→      <div className="flex-1 text-[0.73rem] text-[#374151] leading-relaxed">
   186→        {title && <span className="font-bold text-[#1F2937]">{title} · </span>}
   187→        {children}
   188→      </div>
   189→    </div>
   190→  );
   191→}
   192→
   193→/** A highlighted callout box. */
   194→function Callout({ icon, color, children }: { icon: string; color: string; children: React.ReactNode }) {
   195→  return (
   196→    <div className="mt-3 rounded-xl p-3 flex items-start gap-2.5" style={{ background: `${color}12`, border: `1px solid ${color}33` }}>
   197→      <i className={`fas ${icon} text-[0.8rem] mt-0.5 shrink-0`} style={{ color }} />
   198→      <div className="text-[0.7rem] text-[#374151] leading-relaxed">{children}</div>
   199→    </div>
   200→  );
   201→}
   202→
   203→/** A simple key/value badge row (used for payment methods, levels, etc.). */
   204→function StatRow({ label, value, valueColor }: { label: string; value: string; valueColor?: string }) {
   205→  return (
   206→    <div className="flex items-center justify-between py-1.5 text-[0.72rem]">
   207→      <span className="text-[#6B7280]">{label}</span>
   208→      <span className="font-bold" style={{ color: valueColor || '#1F2937' }}>{value}</span>
   209→    </div>
   210→  );
   211→}
