    10→class ErrorBoundary extends Component<{ children: React.ReactNode }, { hasError: boolean; error?: string }> {
    11→  state = { hasError: false, error: '' };
    12→  static getDerivedStateFromError(e: Error) { return { hasError: true, error: e.message }; }
    13→  render() {
    14→    if (this.state.hasError) {
    15→      return (
    16→        <div className="bg-gradient-to-b from-[#F8F9FA] to-[#F1F5F9] min-h-screen flex items-center justify-center p-6">
    17→          <div className="text-center max-w-[320px]">
    18→            <div className="w-16 h-16 rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mx-auto mb-4">
    19→              <i className="fas fa-exclamation-triangle text-[#F59E0B] text-2xl"></i>
    20→            </div>
    21→            <h2 className="text-lg font-bold text-[#1F2937] mb-2">Oups !</h2>
    22→            <p className="text-sm text-[rgba(0,0,0,0.5)] mb-4">Une erreur inattendue s&apos;est produite.</p>
    23→            <button onClick={() => { this.setState({ hasError: false }); window.location.reload(); }} className="px-6 py-3 rounded-xl bg-[#22C55E] text-white font-semibold border-none cursor-pointer">
    24→              Réessayer
    25→            </button>
    26→          </div>
    27→        </div>
    28→      );
    29→    }
    30→    return this.props.children;
    31→  }
    32→}

    83→// ==================== AUTH SCREEN ====================
    84→function AuthScreen() {
    85→  const { user, setUser, setPage, addToast } = useAppStore();
    86→  // Referral code from URL (lazy init before useState)
    87→  const { prefilledReferral, initialMode } = (() => {
    88→    if (typeof window === 'undefined') return { prefilledReferral: '', initialMode: 'login' as const };
    89→    try {
    90→      const params = new URLSearchParams(window.location.search);
    91→      const ref = params.get('ref');
    92→      if (ref) return { prefilledReferral: ref.toUpperCase(), initialMode: 'register' as const };
    93→    } catch { /* */ }
    94→    return { prefilledReferral: '', initialMode: 'login' as const };
    95→  })();
    96→  const [mode, setMode] = useState<'login' | 'register'>(initialMode);
    97→  const [loading, setLoading] = useState(false);
    98→  const [showPw, setShowPw] = useState({ l: false, r: false, r2: false });
    99→  const [errors, setErrors] = useState<Record<string, string>>({});
   100→  // OTP state (registration only)
   101→  const [otpStep, setOtpStep] = useState(false);
   102→  const [otpEmail, setOtpEmail] = useState('');
   103→  const [otpCode, setOtpCode] = useState('');
   104→  const [otpLoading, setOtpLoading] = useState(false);
   105→  const [simCode, setSimCode] = useState('');
   106→
   107→  if (user) return null;
   108→
   109→  const handleLogin = async (e: React.FormEvent) => {
   110→    e.preventDefault();
   111→    setLoading(true);
   112→    try {
   113→      const fd = new FormData(e.target as HTMLFormElement);
   114→      const email = fd.get('email') as string;
   115→      const password = fd.get('password') as string;
   116→      const res = await fetch('/api/auth/login', { method: 'POST', body: JSON.stringify({ email, password }), headers: { 'Content-Type': 'application/json' } });
   117→      const data = await res.json();
   118→      if (data.success) {
   119→        setUser(data.user); addToast('Bienvenue, ' + data.user.name, 'success'); setPage('videos');
   120→      } else {
   121→        addToast(data.error, 'error');
   122→      }
   123→    } catch { addToast('Erreur réseau', 'error'); }
   124→    setLoading(false);
   125→  };
   126→
   127→  const handleOtpVerify = async () => {
   128→    if (!otpCode || otpCode.length < 6) { addToast('Entrez le code à 6 chiffres', 'error'); return; }
   129→    setOtpLoading(true);
   130→    try {
   131→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'verify', email: otpEmail, code: otpCode, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   132→      const data = await res.json();
   133→      if (data.success && data.user) {
   134→        setUser(data.user);
   135→        addToast('Email vérifié ! Bienvenue, ' + data.user.name, 'success');
   136→        setPage('videos');
   137→      } else {
   138→        addToast(data.error || 'Code invalide', 'error');
   139→      }
   140→    } catch { addToast('Erreur réseau', 'error'); }
   141→    setOtpLoading(false);
   142→  };
   143→
   144→  const handleResendOtp = async () => {
   145→    setOtpLoading(true);
   146→    try {
   147→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'send', email: otpEmail, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   148→      const data = await res.json();
   149→      if (data.success) {
   150→        if (data.plain_code) {
   151→          setSimCode(data.plain_code);
   152→          addToast('Code: ' + data.plain_code, 'info');
   153→        } else {
   154→          addToast('Code renvoyé à ' + otpEmail, 'success');
   155→        }
   156→      } else { addToast(data.error, 'error'); }
   157→    } catch { addToast('Erreur réseau', 'error'); }
   158→    setOtpLoading(false);
   159→  };
   160→
   161→  const handleRegister = async (e: React.FormEvent) => {
   162→    e.preventDefault();
   163→    const fd = new FormData(e.target as HTMLFormElement);
   164→    const name = (fd.get('name') as string)?.trim() || '';
   165→    const email = (fd.get('email') as string)?.trim() || '';
   166→    const password = fd.get('password') as string || '';
   167→    const password2 = fd.get('password2') as string || '';
   168→    const referralCode = (fd.get('referralCode') as string)?.trim().toUpperCase() || '';
   169→    const errs: Record<string, string> = {};
   170→    if (name.length < 2) errs.name = 'Min. 2 caractères';
   171→    if (password.length < 6) errs.password = 'Min. 6 caractères';
   172→    if (password !== password2) errs.password2 = 'Ne correspond pas';
   173→    if (Object.keys(errs).length) { setErrors(errs); return; }
   174→    setErrors({});
   175→    setLoading(true);
   176→    try {
   177→      const res = await fetch('/api/auth/register', { method: 'POST', body: JSON.stringify({ name, email, password, password2, referralCode }), headers: { 'Content-Type': 'application/json' } });
   178→      const data = await res.json();
   179→      if (data.success && data.requires_verification) {
   180→        // Account created — must verify email via OTP
   181→        setOtpEmail(email);
   182→        setOtpStep(true);
   183→        if (data.plain_code) {
   184→          setSimCode(data.plain_code);
   185→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   186→        } else {
   187→          addToast('Code de vérification envoyé à ' + email, 'success');
   188→        }
   189→      } else if (data.success) {
   190→        setUser(data.user); addToast('Compte créé !', 'success'); setPage('videos');
   191→      } else {
   192→        addToast(data.error, 'error');
   193→      }
   194→    } catch { addToast('Erreur réseau', 'error'); }
   195→    setLoading(false);
   196→  };
   197→
   198→  return (
   199→    <section className="absolute inset-0 bg-gradient-to-br from-[#F0FDF4] via-[#F8F9FA] to-[#ECFDF5] flex flex-col items-center justify-center z-[200]">
   200→      {/* Floating decorative orbs */}
   201→      <div className="absolute w-[200px] h-[200px] rounded-full bg-[rgba(34,197,94,0.06)] blur-[80px] top-[15%] left-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite' }} />
   202→      <div className="absolute w-[160px] h-[160px] rounded-full bg-[rgba(20,184,166,0.05)] blur-[80px] bottom-[15%] right-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite 3s reverse' }} />
   203→      <div className="absolute w-[120px] h-[120px] rounded-full bg-[rgba(245,158,11,0.04)] blur-[80px] top-[50%] right-[25%]" style={{ animation: 'orbFloat 7s ease-in-out infinite 1.5s' }} />
   204→
   205→      <div className="w-full max-w-[330px] text-center px-5 relative z-[1]">
   206→        <LogoImg className="w-[100px] h-[100px] mx-auto mb-4" style={{ filter: 'drop-shadow(0 8px 32px rgba(34,197,94,0.25))' }} />
   207→        <h1 className="text-[1.8rem] font-black mb-1 bg-gradient-to-r from-[#22C55E] to-[#16A34A] bg-clip-text text-transparent tracking-[2px]">BE RICH</h1>
   208→        {/* Communication platform tagline */}
   209→        <div className="inline-flex items-center gap-1.5 mb-1 px-3 py-1 rounded-full bg-[rgba(20,184,166,0.1)] border border-[rgba(20,184,166,0.2)]">
   210→          <i className="fas fa-globe-asia text-[#14B8A6] text-[0.6rem]"></i>
   211→          <span className="text-[0.58rem] font-bold text-[#0F766E] uppercase tracking-wide">Plateforme de communication des grandes entreprises</span>
   212→        </div>
   213→        <p className="text-[0.6rem] text-[rgba(0,0,0,0.45)] mb-4">Regardez des vidéos d&rsquo;entreprises chinoises, japonaises et indiennes &mdash; soyez pay&eacute;s&nbsp;!</p>
   214→
   215→        {/* OTP Verification Step — Registration only */}
   216→        {otpStep ? (
   217→          <>
   218→            <div className="w-14 h-14 mx-auto rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mb-4 mt-4">
   219→              <i className="fas fa-shield-alt text-[#22C55E] text-[1.3rem]"></i>
   220→            </div>
   221→            <h2 className="text-[1rem] font-bold text-[#1F2937] mb-1">Vérification email</h2>
   222→            <p className="text-[rgba(0,0,0,0.45)] text-[0.72rem] mb-5">
   223→              Un code a été envoyé à <strong className="text-[#1F2937]">{otpEmail}</strong> pour vérifier votre adresse email
   224→            </p>
   225→
   226→            {/* Simulation mode code display */}
   227→            {simCode && (
   228→              <div className="mb-4 p-4 bg-[rgba(245,158,11,0.12)] border-2 border-[rgba(245,158,11,0.4)] rounded-xl">
   229→                <div className="flex items-center gap-1.5 mb-2">
   230→                  <i className="fas fa-key text-[#F59E0B] text-[0.7rem]"></i>
   231→                  <div className="text-[0.72rem] text-[#D97706] font-bold">Votre code de vérification</div>
   232→                </div>
   233→                <div className="text-[2rem] font-black text-[#F59E0B] tracking-[8px] font-mono text-center py-2">{simCode}</div>
   234→                <div className="flex gap-2 mt-2">
   235→                  <button
   236→                    onClick={() => { setOtpCode(simCode); addToast('Code rempli automatiquement', 'success'); }}
   237→                    className="flex-1 py-2 rounded-lg bg-[#F59E0B] text-white text-[0.75rem] font-bold border-none cursor-pointer flex items-center justify-center gap-1"

   240→                  </button>
   241→                  <button

   243→                    className="py-2 px-3 rounded-lg bg-[rgba(245,158,11,0.15)] text-[0.72rem] text-[#D97706] font-semibold border border-[rgba(245,158,11,0.3)] cursor-pointer"

   279→                    e.preventDefault();
   280→                    const paste = (e.clipboardData.getData('text') || '').replace(/\D/g, '').slice(0, 6);
   281→                    setOtpCode(paste);

   285→            </div>
   286→
   287→            <button
   288→              onClick={handleOtpVerify}
   289→              disabled={otpLoading || otpCode.length < 6}
   290→              className="w-full py-3.5 rounded-xl btn-gradient-green text-[0.88rem] cursor-pointer transition-transform active:scale-[0.97] disabled:opacity-60 flex items-center justify-center gap-2 mb-3"
   291→            >
   292→              {otpLoading ? <div className="w-4 h-4 border-2 border-[rgba(255,255,255,0.3)] border-t-white rounded-full" style={{ animation: 'spin 0.6s linear infinite' }} /> : <><i className="fas fa-check-circle"></i> Vérifier</>}
   293→            </button>
   294→
   295→            <button
   296→              onClick={handleResendOtp}
   297→              disabled={otpLoading}
   298→              className="w-full py-3 rounded-xl border-[1.5px] border-[rgba(0,0,0,0.08)] bg-transparent text-[rgba(0,0,0,0.5)] font-semibold text-[0.82rem] cursor-pointer transition-transform active:scale-95 mb-3"
   299→            >
   300→              <i className="fas fa-redo mr-1.5 text-[0.7rem]"></i>Renvoyer le code

   303→            <button
   304→              onClick={() => { setOtpStep(false); setOtpCode(''); setSimCode(''); }}
   305→              className="text-[0.75rem] text-[rgba(0,0,0,0.4)] cursor-pointer bg-transparent border-none font-medium"
   306→            >
   307→              <i className="fas fa-arrow-left mr-1"></i>Retour
   308→            </button>
   309→          </>
   310→        ) : (
   311→          <>
   312→            <p className="text-[rgba(0,0,0,0.35)] text-[0.72rem] mb-6">{mode === 'login' ? 'Connectez-vous à votre compte.' : 'Rejoignez Be Rich.'}</p>
   313→            <div className="flex bg-[#FFFFFF] rounded-xl p-[3px] mb-6 border border-[rgba(0,0,0,0.08)]">
   314→              <button onClick={() => { setMode('login'); setErrors({}); }} className={`flex-1 py-[11px] text-center text-[0.82rem] font-semibold rounded-lg transition-all border-none cursor-pointer ${mode === 'login' ? 'bg-[#22C55E] text-white shadow-lg' : 'text-[rgba(0,0,0,0.35)]'}`}>Connexion</button>
   315→              <button onClick={() => { setMode('register'); setErrors({}); }} className={`flex-1 py-[11px] text-center text-[0.82rem] font-semibold rounded-lg transition-all border-none cursor-pointer ${mode === 'register' ? 'bg-[#22C55E] text-white shadow-lg' : 'text-[rgba(0,0,0,0.35)]'}`}>Inscription</button>
   316→            </div>
   317→            {mode === 'login' ? (
   318→              <form onSubmit={handleLogin}>
   319→                <div className="mb-4 w-full"><label className="block mb-1.5 text-[0.75rem] font-semibold text-[rgba(0,0,0,0.45)]">Email</label><input name="email" type="email" required placeholder="votre@email.com" className="w-full premium-input" /></div>
   320→                <div className="mb-4 w-full relative"><label className="block mb-1.5 text-[0.75rem] font-semibold text-[rgba(0,0,0,0.45)]">Mot de passe</label><input name="password" type={showPw.l ? 'text' : 'password'} required placeholder="••••••••" className="w-full premium-input pr-11" /><button type="button" onClick={() => setShowPw({ ...showPw, l: !showPw.l })} className="absolute right-3 top-[38px] bg-transparent border-none text-[rgba(0,0,0,0.35)] cursor-pointer p-0.5"><i className={`fas ${showPw.l ? 'fa-eye-slash' : 'fa-eye'}`}></i></button></div>
   321→                <button type="submit" disabled={loading} className="w-full py-3.5 rounded-xl btn-gradient-green text-[0.88rem] cursor-pointer transition-transform active:scale-[0.97] disabled:opacity-60 flex items-center justify-center gap-2">{loading ? <div className="w-4 h-4 border-2 border-[rgba(255,255,255,0.3)] border-t-white rounded-full" style={{ animation: 'spin 0.6s linear infinite' }} /> : <><i className="fas fa-arrow-right"></i> Se connecter</>}</button>
