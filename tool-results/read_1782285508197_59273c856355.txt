   813→function BottomNav() {
   814→  const { currentPage, setPage } = useAppStore();
   815→  const tabs = [
   816→    { id: 'videos', icon: 'fa-video', label: 'Vidéos' },
   817→    { id: 'home', icon: 'fa-coins', label: 'Make Money' },
   818→    { id: 'guide', icon: 'fa-compass', label: 'Guide' },
   819→    { id: 'profile', icon: 'fa-user', label: 'Profil' },
   820→  ];

   822→    if (tabId === 'home') return ['home', 'finance', 'invest', 'game', 'enterprise', 'wallet', 'deposit', 'withdraw'].includes(currentPage);
   823→    if (tabId === 'guide') return currentPage === 'guide';
   824→    return currentPage === tabId;
   825→  };
   826→  return (
   827→    <nav className="h-[60px] bg-white/90 backdrop-blur-xl border-t border-[rgba(0,0,0,0.06)] flex items-center justify-around px-2 shrink-0 safe-area-bottom">
   828→      {tabs.map(t => (
   829→        <button key={t.id} onClick={() => setPage(t.id)} className={`flex flex-col items-center justify-center py-1.5 px-2 border-none cursor-pointer transition-all relative ${isActive(t.id) ? 'text-[#22C55E]' : 'text-[rgba(0,0,0,0.3)]'}`}>
   830→          {isActive(t.id) && <div className="absolute -top-0.5 w-5 h-[3px] rounded-full bg-[#22C55E]"></div>}
   831→          <i className={`fas ${t.icon} text-[0.95rem] mb-0.5`}></i>
   832→          <span className={`text-[0.55rem] ${isActive(t.id) ? 'font-bold text-[#22C55E]' : 'font-semibold'}`}>{t.label}</span>
   833→        </button>
   834→      ))}
   835→    </nav>
   836→  );
   837→}

   840→function ServiceWorkerRegistrar() {
   841→  useEffect(() => {
   842→    if ('serviceWorker' in navigator) {
   843→      navigator.serviceWorker.register('/sw.js').catch(() => {
   844→        // SW registration failed silently
   845→      });
   846→    }
   847→  }, []);
   848→  return null;
   849→}
