    10→class ErrorBoundary extends Component<{ children: React.ReactNode }, { hasError: boolean; error?: string }> {
    11→  state = { hasError: false, error: '' };
    12→  static getDerivedStateFromError(e: Error) { return { hasError: true, error: e.message }; }
    13→  render() {
    14→    if (this.state.hasError) {
    15→      return (
    16→        <div className="bg-gradient-to-b from-[#F8F9FA] to-[#F1F5F9] min-h-screen flex items-center justify-center p-6">
    17→          <div className="text-center max-w-[320px]">
    18→            <div className="w-16 h-16 rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mx-auto mb-4">
    19→              <i className="fas fa-exclamation-triangle text-[#F59E0B] text-2xl"></i>
    20→            </div>
    21→            <h2 className="text-lg font-bold text-[#1F2937] mb-2">Oups !</h2>
    22→            <p className="text-sm text-[rgba(0,0,0,0.5)] mb-4">Une erreur inattendue s&apos;est produite.</p>
    23→            <button onClick={() => { this.setState({ hasError: false }); window.location.reload(); }} className="px-6 py-3 rounded-xl bg-[#22C55E] text-white font-semibold border-none cursor-pointer">
    24→              Réessayer
    25→            </button>
    26→          </div>
    27→        </div>
    28→      );
    29→    }
    30→    return this.props.children;
    31→  }
    32→}

    84→// ==================== AUTH SCREEN ====================
    85→function AuthScreen() {
    86→  const { user, setUser, setPage, addToast } = useAppStore();
    87→  // Referral code from URL (lazy init before useState)
    88→  const { prefilledReferral, initialMode } = (() => {
    89→    if (typeof window === 'undefined') return { prefilledReferral: '', initialMode: 'login' as const };
    90→    try {
    91→      const params = new URLSearchParams(window.location.search);
    92→      const ref = params.get('ref');
    93→      if (ref) return { prefilledReferral: ref.toUpperCase(), initialMode: 'register' as const };
    94→    } catch { /* */ }
    95→    return { prefilledReferral: '', initialMode: 'login' as const };
    96→  })();
    97→  const [mode, setMode] = useState<'login' | 'register'>(initialMode);
    98→  const [loading, setLoading] = useState(false);
    99→  const [showPw, setShowPw] = useState({ l: false, r: false, r2: false });
   100→  const [errors, setErrors] = useState<Record<string, string>>({});
   101→  // OTP state (registration only)
   102→  const [otpStep, setOtpStep] = useState(false);
   103→  const [otpEmail, setOtpEmail] = useState('');
   104→  const [otpCode, setOtpCode] = useState('');
   105→  const [otpLoading, setOtpLoading] = useState(false);
   106→  const [simCode, setSimCode] = useState('');
   107→
   108→  if (user) return null;
   109→
   110→  const handleLogin = async (e: React.FormEvent) => {
   111→    e.preventDefault();
   112→    setLoading(true);
   113→    try {
   114→      const fd = new FormData(e.target as HTMLFormElement);
   115→      const email = fd.get('email') as string;
   116→      const password = fd.get('password') as string;
   117→      const res = await fetch('/api/auth/login', { method: 'POST', body: JSON.stringify({ email, password }), headers: { 'Content-Type': 'application/json' } });
   118→      const data = await res.json();
   119→      if (data.success) {
   120→        setUser(data.user); addToast('Bienvenue, ' + data.user.name, 'success'); setPage('videos');
   121→      } else if (data.needs_verification) {
   122→        // Account exists but email isn't verified — switch to OTP screen.
   123→        setOtpEmail(email);
   124→        setOtpStep(true);
   125→        if (data.plain_code) {
   126→          setSimCode(data.plain_code);
   127→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   128→        } else {
   129→          addToast('Code de vérification envoyé à ' + email, 'success');
   130→        }
   131→      } else {
   132→        addToast(data.error, 'error');
   133→      }
   134→    } catch { addToast('Erreur réseau', 'error'); }
   135→    setLoading(false);
   136→  };
   137→
   138→  const handleOtpVerify = async () => {
   139→    if (!otpCode || otpCode.length < 6) { addToast('Entrez le code à 6 chiffres', 'error'); return; }
   140→    setOtpLoading(true);
   141→    try {
   142→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'verify', email: otpEmail, code: otpCode, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   143→      const data = await res.json();
   144→      if (data.success && data.user) {
   145→        setUser(data.user);
   146→        addToast('Email vérifié ! Bienvenue, ' + data.user.name, 'success');
   147→        setPage('videos');
   148→      } else {
   149→        addToast(data.error || 'Code invalide', 'error');
   150→      }
   151→    } catch { addToast('Erreur réseau', 'error'); }
   152→    setOtpLoading(false);
   153→  };
   154→
   155→  const handleResendOtp = async () => {
   156→    setOtpLoading(true);
   157→    try {
   158→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'send', email: otpEmail, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   159→      const data = await res.json();
   160→      if (data.success) {
   161→        if (data.plain_code) {
   162→          setSimCode(data.plain_code);
   163→          addToast('Code: ' + data.plain_code, 'info');
   164→        } else {
   165→          addToast('Code renvoyé à ' + otpEmail, 'success');
   166→        }
   167→      } else { addToast(data.error, 'error'); }
   168→    } catch { addToast('Erreur réseau', 'error'); }
   169→    setOtpLoading(false);
   170→  };
   171→
   172→  const handleRegister = async (e: React.FormEvent) => {
   173→    e.preventDefault();
   174→    const fd = new FormData(e.target as HTMLFormElement);
   175→    const name = (fd.get('name') as string)?.trim() || '';
   176→    const email = (fd.get('email') as string)?.trim() || '';
   177→    const phone = (fd.get('phone') as string)?.trim() || '';
   178→    const password = fd.get('password') as string || '';
   179→    const password2 = fd.get('password2') as string || '';
   180→    const referralCode = (fd.get('referralCode') as string)?.trim().toUpperCase() || '';
   181→    const errs: Record<string, string> = {};
   182→    if (name.length < 2) errs.name = 'Min. 2 caractères';
   183→    if (password.length < 6) errs.password = 'Min. 6 caractères';
   184→    if (password !== password2) errs.password2 = 'Ne correspond pas';
   185→    if (Object.keys(errs).length) { setErrors(errs); return; }
   186→    setErrors({});
   187→    setLoading(true);
   188→    try {
   189→      const res = await fetch('/api/auth/register', { method: 'POST', body: JSON.stringify({ name, email, phone: phone.trim(), password, password2, referralCode }), headers: { 'Content-Type': 'application/json' } });
   190→      const data = await res.json();
   191→      if (data.success && data.requires_verification) {
   192→        // Account created — must verify email via OTP
   193→        setOtpEmail(email);
   194→        setOtpStep(true);
   195→        if (data.plain_code) {
   196→          setSimCode(data.plain_code);
   197→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   198→        } else {
   199→          addToast('Code de vérification envoyé à ' + email, 'success');
   200→        }
   201→      } else if (data.success) {
   202→        setUser(data.user); addToast('Compte créé !', 'success'); setPage('videos');
   203→      } else {
   204→        addToast(data.error, 'error');
   205→      }
   206→    } catch { addToast('Erreur réseau', 'error'); }
   207→    setLoading(false);
   208→  };
   209→
   210→  return (
   211→    <section className="absolute inset-0 bg-gradient-to-br from-[#F0FDF4] via-[#F8F9FA] to-[#ECFDF5] flex flex-col items-center justify-start sm:justify-center overflow-y-auto z-[200]">
   212→      {/* Floating decorative orbs */}
   213→      <div className="absolute w-[200px] h-[200px] rounded-full bg-[rgba(34,197,94,0.06)] blur-[80px] top-[15%] left-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite' }} />
   214→      <div className="absolute w-[160px] h-[160px] rounded-full bg-[rgba(20,184,166,0.05)] blur-[80px] bottom-[15%] right-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite 3s reverse' }} />
   215→      <div className="absolute w-[120px] h-[120px] rounded-full bg-[rgba(245,158,11,0.04)] blur-[80px] top-[50%] right-[25%]" style={{ animation: 'orbFloat 7s ease-in-out infinite 1.5s' }} />
   216→
   217→      <div className="w-full max-w-[330px] text-center px-5 relative z-[1] py-6 my-auto">
   218→        <LogoImg className="w-[100px] h-[100px] mx-auto mb-4" style={{ filter: 'drop-shadow(0 8px 32px rgba(34,197,94,0.25))' }} />
   219→        <h1 className="text-[1.8rem] font-black mb-1 bg-gradient-to-r from-[#22C55E] to-[#16A34A] bg-clip-text text-transparent tracking-[2px]">BE RICH</h1>
   220→        {/* Communication platform tagline */}
   221→        <div className="inline-flex items-center gap-1.5 mb-1 px-3 py-1 rounded-full bg-[rgba(20,184,166,0.1)] border border-[rgba(20,184,166,0.2)]">
   222→          <i className="fas fa-globe-asia text-[#14B8A6] text-[0.6rem]"></i>

   226→
   227→        {/* OTP Verification Step — Registration only */}
   228→        {otpStep ? (
   229→          <>
   230→            <div className="w-14 h-14 mx-auto rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mb-4 mt-4">

   871→// ==================== BOTTOM NAV ====================
   872→function BottomNav() {
   873→  const { currentPage, setPage } = useAppStore();
   874→  const tabs = [
   875→    { id: 'videos', icon: 'fa-video', label: 'Vidéos' },
   876→    { id: 'home', icon: 'fa-coins', label: 'Make Money' },
   877→    { id: 'guide', icon: 'fa-compass', label: 'Guide' },
   878→    { id: 'profile', icon: 'fa-user', label: 'Profil' },
   879→  ];
   880→  const isActive = (tabId: string) => {
   881→    if (tabId === 'home') return ['home', 'finance', 'invest', 'game', 'enterprise', 'wallet', 'deposit', 'withdraw'].includes(currentPage);
   882→    if (tabId === 'guide') return currentPage === 'guide';
   883→    return currentPage === tabId;
   884→  };
   885→  return (
   886→    <nav className="h-[60px] bg-white/90 backdrop-blur-xl border-t border-[rgba(0,0,0,0.06)] flex items-center justify-around px-2 shrink-0 safe-area-bottom">
   887→      {tabs.map(t => (
   888→        <button key={t.id} onClick={() => setPage(t.id)} className={`flex flex-col items-center justify-center py-1.5 px-2 border-none cursor-pointer transition-all relative ${isActive(t.id) ? 'text-[#22C55E]' : 'text-[rgba(0,0,0,0.3)]'}`}>
   889→          {isActive(t.id) && <div className="absolute -top-0.5 w-5 h-[3px] rounded-full bg-[#22C55E]"></div>}
   890→          <i className={`fas ${t.icon} text-[0.95rem] mb-0.5`}></i>
   891→          <span className={`text-[0.55rem] ${isActive(t.id) ? 'font-bold text-[#22C55E]' : 'font-semibold'}`}>{t.label}</span>
   892→        </button>
   893→      ))}
   894→    </nav>
   895→  );
   896→}

   899→function ServiceWorkerRegistrar() {
   900→  useEffect(() => {
   901→    if ('serviceWorker' in navigator) {
   902→      navigator.serviceWorker.register('/sw.js').catch(() => {
   903→        // SW registration failed silently
   904→      });
   905→    }
   906→  }, []);
   907→  return null;
