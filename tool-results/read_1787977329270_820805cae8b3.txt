    25→// === Drag state helpers ===
    26→const POS_STORAGE_KEY = 'beRich.floatingGift.pos';
    27→const BUTTON_SIZE = 64; // main circle width/height
    28→const DEFAULT_MARGIN_RIGHT = 18;
    29→const DEFAULT_MARGIN_BOTTOM = 80;
    30→const EDGE_PAD = 4; // px — keep button inside viewport
    31→
    32→interface Pos { x: number; y: number; }
    33→
    34→function getDefaultPos(): Pos {
    35→  if (typeof window === 'undefined') return { x: 0, y: 0 };
    36→  return {
    37→    x: window.innerWidth - BUTTON_SIZE - DEFAULT_MARGIN_RIGHT,
    38→    y: window.innerHeight - BUTTON_SIZE - DEFAULT_MARGIN_BOTTOM,
    39→  };
    40→}
    41→
    42→function clampPos(p: Pos): Pos {
    43→  if (typeof window === 'undefined') return p;
    44→  const maxX = Math.max(EDGE_PAD, window.innerWidth - BUTTON_SIZE - EDGE_PAD);
    45→  const maxY = Math.max(EDGE_PAD, window.innerHeight - BUTTON_SIZE - EDGE_PAD);
    46→  return {
    47→    x: Math.min(Math.max(EDGE_PAD, p.x), maxX),
    48→    y: Math.min(Math.max(EDGE_PAD, p.y), maxY),
    49→  };
    50→}
    51→
    52→function loadStoredPos(): Pos {
    53→  try {
    54→    const raw = typeof window !== 'undefined' ? window.localStorage.getItem(POS_STORAGE_KEY) : null;
    55→    if (raw) {
    56→      const p = JSON.parse(raw);
    57→      if (typeof p?.x === 'number' && typeof p?.y === 'number') {
    58→        return clampPos({ x: p.x, y: p.y });
    59→      }
    60→    }
    61→  } catch { /* ignore */ }
    62→  return getDefaultPos();
    63→}
    64→
    65→function saveStoredPos(p: Pos) {
    66→  try { window.localStorage.setItem(POS_STORAGE_KEY, JSON.stringify(p)); } catch { /* ignore */ }
    67→}
