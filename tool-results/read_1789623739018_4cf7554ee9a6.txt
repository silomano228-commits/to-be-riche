    86→    86→function AuthScreen() {

    88→    88→  // Referral code from URL (lazy init before useState)
    89→    89→  const { prefilledReferral, initialMode } = (() => {
    90→    90→    if (typeof window === 'undefined') return { prefilledReferral: '', initialMode: 'login' as const };
    91→    91→    try {
    92→    92→      const params = new URLSearchParams(window.location.search);
    93→    93→      const ref = params.get('ref');
    94→    94→      if (ref) return { prefilledReferral: ref.toUpperCase(), initialMode: 'register' as const };
    95→    95→    } catch { /* */ }
    96→    96→    return { prefilledReferral: '', initialMode: 'login' as const };
    97→    97→  })();
    98→    98→  const [mode, setMode] = useState<'login' | 'register'>(initialMode);
    99→    99→  const [loading, setLoading] = useState(false);
   100→   100→  const [showPw, setShowPw] = useState({ l: false, r: false, r2: false });
   101→   101→  const [errors, setErrors] = useState<Record<string, string>>({});
   102→   102→  // OTP state (registration only)
   103→   103→  const [otpStep, setOtpStep] = useState(false);
   104→   104→  const [otpEmail, setOtpEmail] = useState('');
   105→   105→  const [otpCode, setOtpCode] = useState('');
   106→   106→  const [otpLoading, setOtpLoading] = useState(false);
   107→   107→  const [simCode, setSimCode] = useState('');

   111→   111→  const handleLogin = async (e: React.FormEvent) => {
   112→   112→    e.preventDefault();
   113→   113→    setLoading(true);
   114→   114→    try {
   115→   115→      const fd = new FormData(e.target as HTMLFormElement);
   116→   116→      const email = fd.get('email') as string;
   117→   117→      const password = fd.get('password') as string;
   118→   118→      const res = await fetch('/api/auth/login', { method: 'POST', body: JSON.stringify({ email, password }), headers: { 'Content-Type': 'application/json' } });
   119→   119→      const data = await res.json();
   120→   120→      if (data.success) {
   121→   121→        setUser(data.user); addToast('Bienvenue, ' + data.user.name, 'success'); setPage('videos');
   122→   122→      } else if (data.needs_verification) {
   123→   123→        // Account exists but email isn't verified — switch to OTP screen.
   124→   124→        setOtpEmail(email);
   125→   125→        setOtpStep(true);
   126→   126→        if (data.plain_code) {
   127→   127→          setSimCode(data.plain_code);
   128→   128→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   129→   129→        } else {
   130→   130→          addToast('Code de vérification envoyé à ' + email, 'success');
   131→   131→        }
   132→   132→      } else {
   133→   133→        addToast(data.error, 'error');
   134→   134→      }
   135→   135→    } catch { addToast('Erreur réseau', 'error'); }
   136→   136→    setLoading(false);
   137→   137→  };
   138→   138→
   139→   139→  const handleOtpVerify = async () => {

   144→   144→      const data = await res.json();

   146→   146→        setUser(data.user);
   147→   147→        addToast('Email vérifié ! Bienvenue, ' + data.user.name, 'success');
   148→   148→        setPage('videos');
   149→   149→      } else {
   150→   150→        addToast(data.error || 'Code invalide', 'error');
   151→   151→      }
   152→   152→    } catch { addToast('Erreur réseau', 'error'); }
   153→   153→    setOtpLoading(false);
   154→   154→  };
   155→   155→
   156→   156→  const handleResendOtp = async () => {
