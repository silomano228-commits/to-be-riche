   508→function VideoPlayerModal({ video, onClose, onReward }: {
   509→  video: VideoItem;
   510→  onClose: () => void;
   511→  onReward: (reward: number) => Promise<void>;
   512→}) {
   513→  const [watchedPercent, setWatchedPercent] = useState(0);
   514→  const [canClaim, setCanClaim] = useState(false);
   515→  const [claiming, setClaiming] = useState(false);
   516→  const [claimed, setClaimed] = useState(false);
   517→  const [error, setError] = useState<string | null>(null);
   518→  const [iframeLoaded, setIframeLoaded] = useState(false);
   519→  const startTimeRef = useRef<number>(Date.now());
   520→  const claimedRef = useRef(false);
   521→
   522→  // Total video duration in seconds (from catalog). Used for time-based progress.
   523→  const totalSeconds = video.durationMin * 60;
   524→  const requiredSeconds = totalSeconds * (MIN_WATCH_PCT / 100);
   525→
   526→  // Time-based progress tracker: increments watchedPercent based on elapsed time.
   527→  // This is more reliable than the YouTube IFrame API's getCurrentTime(), which
   528→  // can fail to report when the API fires spurious onError events.
   529→  useEffect(() => {
   530→    startTimeRef.current = Date.now();
   531→    const interval = setInterval(() => {
   532→      const elapsed = (Date.now() - startTimeRef.current) / 1000;
   533→      const pct = Math.min(100, (elapsed / totalSeconds) * 100);
   534→      setWatchedPercent(pct);
   535→      if (pct >= MIN_WATCH_PCT) setCanClaim(true);
   536→    }, 1000);
   537→    return () => clearInterval(interval);
   538→  }, [video.id, totalSeconds]);
   539→
   540→  // If the iframe doesn't load within 15 seconds, show a fallback error.
   541→  useEffect(() => {
   542→    const timeout = setTimeout(() => {
   543→      if (!iframeLoaded) {
   544→        setError('La vidéo met trop de temps à charger. Vérifiez votre connexion ou essayez une autre vidéo.');
   545→      }
   546→    }, 15000);
   547→    return () => clearTimeout(timeout);
   548→  }, [iframeLoaded]);
   549→
   550→  const preventScroll = (e: React.WheelEvent | React.TouchEvent) => {
   551→    e.preventDefault();
   552→  };
   553→
   554→  const handleClaim = async () => {
   555→    if (claiming || claimedRef.current) return;
   556→    claimedRef.current = true;
   557→    setClaiming(true);
   558→    setError(null);
   559→    try {
   560→      const pct = Math.max(watchedPercent, MIN_WATCH_PCT);
   561→      const res = await authFetch('/api/videos/reward', {
   562→        method: 'POST',
   563→        headers: { 'Content-Type': 'application/json' },
   564→        body: JSON.stringify({ videoId: video.id, watchedPercent: pct }),
   565→      });
   566→      const data = await res.json();
   567→      if (data.success) {
   568→        setClaimed(true);
   569→      } else {
   570→        setError(data.error || 'Erreur lors de la réclamation.');
   571→        claimedRef.current = false;
   572→      }
   573→    } catch {
   574→      setError('Erreur de connexion. Veuillez réessayer.');
   575→      claimedRef.current = false;
   576→    }
   577→    setClaiming(false);
   578→  };
   579→
   580→  const handleClose = async () => {
   581→    if (claimed) {
   582→      await onReward(video.reward);
   583→    } else {
   584→      onClose();
   585→    }
   586→  };
   587→
   588→
   589→  const remainingPct = Math.max(0, Math.ceil(MIN_WATCH_PCT - watchedPercent));
   590→  const remainingSec = Math.max(0, Math.ceil(requiredSeconds - (Date.now() - startTimeRef.current) / 1000));
   591→
   592→  // Build the YouTube embed URL. Using youtube.com (not nocookie) for maximum
   593→  // compatibility. autoplay=1, controls=0 (hidden), rel=0, modestbranding=1.

   595→
   596→  return (
   597→    <div className="fixed inset-0 z-[9000] flex items-center justify-center p-4" style={{ background: 'rgba(0,0,0,0.92)' }}>
   598→      <div className="w-full max-w-[420px]">
   599→        {/* Header — Quit (X) button ALWAYS visible */}
   600→        <div className="flex items-center justify-between mb-3 gap-2">
   601→          <div className="flex items-center gap-2 min-w-0 flex-1">
   602→            <span className="text-[0.55rem] font-bold px-1.5 py-0.5 rounded text-white flex-shrink-0 flex items-center gap-1" style={{ background: THEME_COLOR[video.category] || '#14B8A6' }}>
   603→              <i className={`${THEME_ICON[video.category] || 'fas fa-briefcase'} text-[0.45rem]`}></i>
   604→              {THEME_LABEL[video.category] || video.category}
   605→            </span>
   606→            <span className="text-[0.6rem] text-white/60 truncate">Sponsorisé par {video.sponsor}</span>
   607→          </div>
   608→          <button
   609→            onClick={claimed ? handleClose : onClose}
   610→            aria-label="Quitter la vidéo"
   611→            title="Quitter la vidéo"
   612→            className="flex-shrink-0 w-11 h-11 rounded-xl flex items-center justify-center cursor-pointer transition-all active:scale-95 border-none"
   613→            style={{ background: 'rgba(239,68,68,0.95)', color: '#FFFFFF', boxShadow: '0 2px 10px rgba(239,68,68,0.45)' }}

   855→// ===== Share modal with native share sheet =====
   856→function ShareModal({ referralCode, addToast, onClose }: { referralCode: string; addToast: (msg: string, type: 'success' | 'error' | 'info') => void; onClose: () => void }) {
   857→  const shareUrl = typeof window !== 'undefined' ? `${window.location.origin}/?ref=${referralCode}` : '';
   858→  const shareText = `Rejoins Be Rich, la plateforme de communication pour les grandes entreprises ! Gagne de l'argent en regardant des vidéos. ${shareUrl}`;
   859→
   860→  const handleNativeShare = async () => {
   861→    if (navigator.share) {
   862→      try {
   863→        await navigator.share({ title: 'Be Rich', text: shareText, url: shareUrl });
   864→        onClose();
   865→      } catch { /* user cancelled */ }
   866→    } else {
   867→      // Fallback: copy full share text to clipboard
   868→      await copyLink(true);
   869→    }
   870→  };
   871→
   872→  const copyLink = async (copyFullText = false) => {
   873→    const textToCopy = copyFullText ? shareText : shareUrl;
