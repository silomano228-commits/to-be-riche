   505→   505→// ===== Video player modal — simple iframe + time-based progress (robust, no YT API dependency) =====
   506→   506→const MIN_WATCH_PCT = 30; // must match /api/videos/reward threshold
   507→   507→
   508→   508→function VideoPlayerModal({ video, onClose, onReward }: {
   509→   509→  video: VideoItem;
   510→   510→  onClose: () => void;
   511→   511→  onReward: (reward: number) => Promise<void>;
   512→   512→}) {
   513→   513→  const [watchedPercent, setWatchedPercent] = useState(0);
   514→   514→  const [canClaim, setCanClaim] = useState(false);
   515→   515→  const [claiming, setClaiming] = useState(false);
   516→   516→  const [claimed, setClaimed] = useState(false);
   517→   517→  const [error, setError] = useState<string | null>(null);
   518→   518→  const [iframeLoaded, setIframeLoaded] = useState(false);
   519→   519→  const startTimeRef = useRef<number>(Date.now());
   520→   520→  const claimedRef = useRef(false);
   521→   521→
   522→   522→  // Total video duration in seconds (from catalog). Used for time-based progress.
   523→   523→  const totalSeconds = video.durationMin * 60;
   524→   524→  const requiredSeconds = totalSeconds * (MIN_WATCH_PCT / 100);
   525→   525→
   526→   526→  // Time-based progress tracker: increments watchedPercent based on elapsed time.
   527→   527→  // This is more reliable than the YouTube IFrame API's getCurrentTime(), which
   528→   528→  // can fail to report when the API fires spurious onError events.
   529→   529→  useEffect(() => {
   530→   530→    startTimeRef.current = Date.now();
   531→   531→    const interval = setInterval(() => {
   532→   532→      const elapsed = (Date.now() - startTimeRef.current) / 1000;
   533→   533→      const pct = Math.min(100, (elapsed / totalSeconds) * 100);
   534→   534→      setWatchedPercent(pct);
   535→   535→      if (pct >= MIN_WATCH_PCT) setCanClaim(true);
   536→   536→    }, 1000);
   537→   537→    return () => clearInterval(interval);
   538→   538→  }, [video.id, totalSeconds]);
   539→   539→
   540→   540→  // If the iframe doesn't load within 15 seconds, show a fallback error.
   541→   541→  useEffect(() => {
   542→   542→    const timeout = setTimeout(() => {
   543→   543→      if (!iframeLoaded) {
   544→   544→        setError('La vidéo met trop de temps à charger. Vérifiez votre connexion ou essayez une autre vidéo.');
   545→   545→      }
   546→   546→    }, 15000);
   547→   547→    return () => clearTimeout(timeout);
   548→   548→  }, [iframeLoaded]);
   549→   549→
   550→   550→  const preventScroll = (e: React.WheelEvent | React.TouchEvent) => {
   551→   551→    e.preventDefault();
   552→   552→  };
   553→   553→
   554→   554→  const handleClaim = async () => {
   555→   555→    if (claiming || claimedRef.current) return;
   556→   556→    claimedRef.current = true;
   557→   557→    setClaiming(true);
   558→   558→    setError(null);
   559→   559→    try {
   560→   560→      const pct = Math.max(watchedPercent, MIN_WATCH_PCT);
   561→   561→      const res = await authFetch('/api/videos/reward', {
   562→   562→        method: 'POST',
   563→   563→        headers: { 'Content-Type': 'application/json' },
   564→   564→        body: JSON.stringify({ videoId: video.id, watchedPercent: pct }),
   565→   565→      });
   566→   566→      const data = await res.json();
   567→   567→      if (data.success) {
   568→   568→        setClaimed(true);
   569→   569→      } else {
   570→   570→        setError(data.error || 'Erreur lors de la réclamation.');
   571→   571→        claimedRef.current = false;
   572→   572→      }
   573→   573→    } catch {
   574→   574→      setError('Erreur de connexion. Veuillez réessayer.');
   575→   575→      claimedRef.current = false;
   576→   576→    }
   577→   577→    setClaiming(false);
   578→   578→  };
   579→   579→
   580→   580→  const handleClose = async () => {
   581→   581→    if (claimed) {
   582→   582→      await onReward(video.reward);
   583→   583→    } else {
   584→   584→      onClose();
   585→   585→    }
   586→   586→  };
   587→   587→
   588→   588→
   589→   589→  const remainingPct = Math.max(0, Math.ceil(MIN_WATCH_PCT - watchedPercent));
   590→   590→  const remainingSec = Math.max(0, Math.ceil(requiredSeconds - (Date.now() - startTimeRef.current) / 1000));
   591→   591→
   592→   592→  // Build the YouTube embed URL. Using youtube.com (not nocookie) for maximum
   593→   593→  // compatibility. autoplay=1, controls=0 (hidden), rel=0, modestbranding=1.
   594→   594→  const embedUrl = `https://www.youtube.com/embed/${video.id}?autoplay=1&controls=0&disablekb=1&modestbranding=1&rel=0&playsinline=1&fs=0&iv_load_policy=3&enablejsapi=0`;
   595→   595→
   596→   596→  return (
   597→   597→    <div className="fixed inset-0 z-[9000] flex items-center justify-center p-4" style={{ background: 'rgba(0,0,0,0.92)' }}>
   598→   598→      <div className="w-full max-w-[420px]">
   599→   599→        {/* Header — Quit (X) button ALWAYS visible */}
   600→   600→        <div className="flex items-center justify-between mb-3 gap-2">
   601→   601→          <div className="flex items-center gap-2 min-w-0 flex-1">
   602→   602→            <span className="text-[0.55rem] font-bold px-1.5 py-0.5 rounded text-white flex-shrink-0 flex items-center gap-1" style={{ background: THEME_COLOR[video.category] || '#14B8A6' }}>
   603→   603→              <i className={`${THEME_ICON[video.category] || 'fas fa-briefcase'} text-[0.45rem]`}></i>
   604→   604→              {THEME_LABEL[video.category] || video.category}
   605→   605→            </span>
   606→   606→            <span className="text-[0.6rem] text-white/60 truncate">Sponsorisé par {video.sponsor}</span>
   607→   607→          </div>
   608→   608→          <button
   609→   609→            onClick={claimed ? handleClose : onClose}
   610→   610→            aria-label="Quitter la vidéo"
   611→   611→            title="Quitter la vidéo"
   612→   612→            className="flex-shrink-0 w-11 h-11 rounded-xl flex items-center justify-center cursor-pointer transition-all active:scale-95 border-none"
   613→   613→            style={{ background: 'rgba(239,68,68,0.95)', color: '#FFFFFF', boxShadow: '0 2px 10px rgba(239,68,68,0.45)' }}
   614→   614→          >
   615→   615→            <i className="fas fa-times text-[1.15rem]"></i>
   616→   616→          </button>
   617→   617→        </div>
   618→   618→
   619→   619→        {/* Title */}
   620→   620→        <h3 className="text-[0.9rem] font-bold text-white mb-2">{video.title}</h3>
   621→   621→
   622→   622→        {/* Player container — simple iframe with pointer-events:none overlay */}
   623→   623→        <div
   624→   624→          onWheel={preventScroll}
   625→   625→          onTouchMove={preventScroll}
   626→   626→          className="relative w-full rounded-xl overflow-hidden bg-black"
   627→   627→          style={{ aspectRatio: '16/9' }}
   628→   628→        >

   633→   633→            style={{ border: 'none', pointerEvents: 'none' }}
