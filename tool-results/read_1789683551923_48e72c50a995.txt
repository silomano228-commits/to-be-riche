    49→export default function MissionsScreen() {
    50→  const { user, setPage, addToast } = useAppStore();
    51→  const [tab, setTab] = useState<'dashboard'|'campaigns'|'myimages'|'eligibility'|'loans'>('dashboard');
    52→  const [dash, setDash] = useState<DashboardData|null>(null);
    53→  const [campaigns, setCampaigns] = useState<Campaign[]>([]);
    54→  const [images, setImages] = useState<MissionImage[]>([]);
    55→  const [loading, setLoading] = useState(true);
    56→  const [gening, setGening] = useState(false);
    57→  const [prompt, setPrompt] = useState('');
    58→  const [selCamp, setSelCamp] = useState<Campaign|null>(null);
    59→  const [showGen, setShowGen] = useState(false);
    60→  const [elig, setElig] = useState<any>(null);
    61→  const [loans, setLoans] = useState<any[]>([]);
    62→
    63→  useEffect(() => { loadData(); }, []);
    64→
    65→  const loadData = async () => {
    66→    setLoading(true);
    67→    try {
    68→      const [dr, cr, ir] = await Promise.all([
    69→        authFetch('/api/missions/dashboard'),
    70→        authFetch('/api/missions/campaigns'),
    71→        authFetch('/api/missions/images?limit=20'),
    72→      ]);
    73→      const dd = await dr.json(); if (dd.success) setDash(dd.dashboard);
    74→      const cd = await cr.json(); if (cd.success) setCampaigns(cd.campaigns);
    75→      const id2 = await ir.json(); if (id2.success) setImages(id2.images);
    76→    } catch {}
    77→    setLoading(false);
    78→  };
    79→
    80→  const loadElig = async () => {
    81→    try { const r = await authFetch('/api/missions/eligibility'); const d = await r.json(); if (d.success) setElig({ ...d.eligibility, userLevel: d.userLevel, userLevelLabel: d.userLevelLabel, activeLoan: d.activeLoan }); } catch {}
    82→  };
    83→
    84→  const loadLoans = async () => {
    85→    try { const r = await authFetch('/api/missions/loans'); const d = await r.json(); if (d.success) setLoans(d.loans); } catch {}
    86→  };
    87→
    88→  const handleGen = async () => {
    89→    if (!selCamp || !prompt.trim()) return;
    90→    setGening(true);
    91→    try {
    92→      const r = await authFetch('/api/missions/images', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ campaignId: selCamp.id, prompt }) });
    93→      const d = await r.json();
    94→      if (d.success) { addToast('Image générée ! Validation IA en cours...', 'success'); setShowGen(false); setPrompt(''); setSelCamp(null); setTimeout(loadData, 3000); }
    95→      else addToast(d.error || 'Erreur', 'error');
    96→    } catch { addToast('Erreur de génération', 'error'); }
    97→    setGening(false);
    98→  };
    99→
   100→  const handleUpload = async (file: File) => {
   101→    if (!selCamp) return;
   102→    setGening(true);
   103→    const reader = new FileReader();
   104→    reader.onload = async (e) => {
   105→      const b64 = (e.target?.result as string).split(',')[1];
   106→      try {
   107→        const r = await authFetch('/api/missions/images', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ campaignId: selCamp.id, imageData: b64 }) });
   108→        const d = await r.json();
   109→        if (d.success) { addToast('Image envoyée ! Validation en cours...', 'success'); setShowGen(false); setSelCamp(null); setTimeout(loadData, 3000); }
   110→        else addToast(d.error || 'Erreur', 'error');
   111→      } catch { addToast('Erreur', 'error'); }
   112→      setGening(false);
   113→    };
   114→    reader.readAsDataURL(file);
   115→  };
   116→
   117→  const reqLoan = async (amt: number) => {
   118→    try {
   119→      const r = await authFetch('/api/missions/loans', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ amountCfa: amt }) });
   120→      const d = await r.json();
   121→      if (d.success) { addToast(`Demande de ${amt} FCFA enregistrée !`, 'success'); loadLoans(); loadElig(); }
   122→      else addToast(d.error || 'Conditions non remplies', 'error');
   123→    } catch { addToast('Erreur', 'error'); }
   124→  };
   125→
   126→  const depositCaution = async () => {
   127→    try {
   128→      const r = await authFetch('/api/missions/caution', { method: 'POST' });
   129→      const d = await r.json();
   130→      if (d.success) { addToast('Caution constituée !', 'success'); loadData(); }
   131→      else addToast(d.error || 'Erreur', 'error');
   132→    } catch { addToast('Erreur', 'error'); }
   133→  };
   134→
   135→  const repay = async (lid: string) => {
   136→    const s = prompt('Montant à rembourser (FCFA) :');
   137→    if (!s) return;
   138→    const amt = parseInt(s);
   139→    if (isNaN(amt) || amt <= 0) { addToast('Montant invalide', 'error'); return; }
   140→    try {
   141→      const r = await authFetch(`/api/missions/loans/${lid}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'repay', amountCfa: amt }) });
   142→      const d = await r.json();
   143→      if (d.success) { addToast('Remboursement effectué !', 'success'); loadLoans(); loadData(); refreshUser(); }
   144→      else addToast(d.error || 'Erreur', 'error');
   145→    } catch { addToast('Erreur', 'error'); }
   146→  };
   147→
   148→  if (!user) return null;
   149→
   150→  const tabs = [
   151→    { id: 'dashboard' as const, icon: 'fa-th-large', label: 'Tableau' },
   152→    { id: 'campaigns' as const, icon: 'fa-bullhorn', label: 'Missions' },
   153→    { id: 'myimages' as const, icon: 'fa-images', label: 'Mes images' },
   154→    { id: 'eligibility' as const, icon: 'fa-chart-bar', label: 'Éligibilité' },
   155→    { id: 'loans' as const, icon: 'fa-hand-holding-usd', label: 'Prêts' },
   156→  ];
   157→
   158→  return (
   159→    <>
   160→      <Header title="Espace Jeunes" rightElement={
   161→        <button onClick={() => setPage('profile')} className="w-9 h-9 rounded-[10px] flex items-center justify-center bg-[rgba(0,0,0,0.04)] text-[#64748B] cursor-pointer border-none text-[0.85rem]"><i className="far fa-user-circle"></i></button>
   162→      } />
   163→      <div className="flex gap-1 px-4 pt-2 pb-1 overflow-x-auto bg-white border-b border-[rgba(0,0,0,0.05)]">
   164→        {tabs.map(t => (
   165→          <button key={t.id} onClick={() => { setTab(t.id); if (t.id === 'eligibility') loadElig(); if (t.id === 'loans') loadLoans(); }}
   166→            className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-[0.72rem] font-semibold border-none cursor-pointer transition-all whitespace-nowrap ${tab === t.id ? 'bg-[#22C55E] text-white' : 'bg-[rgba(0,0,0,0.03)] text-[#64748B]'}`}>
   167→            <i className={`fas ${t.icon} text-[0.65rem]`}></i> {t.label}
   168→          </button>
   169→        ))}
   170→      </div>
   171→      <div className="flex-1 overflow-y-auto px-4 py-4">
   172→        {loading && !dash ? <div className="flex items-center justify-center py-12"><div className="w-6 h-6 border-[2.5px] border-[rgba(0,0,0,0.08)] border-t-[#22C55E] rounded-full" style={{ animation: 'spin 0.7s linear infinite' }} /></div>
   173→        : tab === 'dashboard' ? <DashV dash={dash} go={setTab} />
   174→        : tab === 'campaigns' ? <CampV camps={campaigns} onSel={c => { setSelCamp(c); setShowGen(true); }} />
   175→        : tab === 'myimages' ? <ImgV imgs={images} />
   176→        : tab === 'eligibility' ? <EligV elig={elig} onLoan={reqLoan} onCaution={depositCaution} dash={dash} onLoad={loadElig} />
   177→        : tab === 'loans' ? <LoanV loans={loans} onRepay={repay} />
   178→        : null}
   179→      </div>
   180→
   181→      {showGen && selCamp && (
   182→        <div className="fixed inset-0 z-50 bg-[rgba(0,0,0,0.6)] flex items-end justify-center" onClick={() => setShowGen(false)}>
   183→          <div className="bg-white w-full max-w-lg rounded-t-2xl p-5 max-h-[85vh] overflow-y-auto" onClick={e => e.stopPropagation()}>
   184→            <div className="flex justify-between items-center mb-4">
   185→              <h3 className="text-lg font-bold text-[#1F2937]">Générer une image</h3>
   186→              <button onClick={() => setShowGen(false)} className="text-[#94A3B8] text-xl cursor-pointer border-none bg-transparent"><i className="fas fa-times"></i></button>
   187→            </div>
   188→            <div className="mb-3 p-3 bg-[rgba(34,197,94,0.06)] rounded-xl">
   189→              <div className="text-[0.72rem] font-bold text-[#22C55E] mb-1">{selCamp.brand} — {selCamp.name}</div>
   190→              <div className="text-[0.65rem] text-[#64748B] leading-relaxed">{selCamp.brief}</div>
   191→              {selCamp.constraints && <div className="text-[0.6rem] text-[#94A3B8] mt-1">Contraintes: {selCamp.constraints}</div>}
   192→            </div>
   193→            <div className="mb-3">
   194→              <label className="text-[0.72rem] font-semibold text-[#1F2937] mb-1.5 block">Décrivez votre image</label>
   195→              <textarea value={prompt} onChange={e => setPrompt(e.target.value)} placeholder="Ex: Une voiture Mercedes dans une rue moderne au coucher du soleil..." className="w-full p-3 rounded-xl border border-[rgba(0,0,0,0.1)] text-[0.78rem] resize-none h-24 focus:outline-none focus:border-[#22C55E]" />
   196→            </div>
   197→            <button onClick={handleGen} disabled={gening || !prompt.trim()} className="w-full py-3 rounded-xl bg-[#22C55E] text-white font-semibold text-[0.82rem] border-none cursor-pointer disabled:opacity-50 flex items-center justify-center gap-2 mb-3">
   198→              {gening ? <><div className="w-4 h-4 border-[2px] border-white/30 border-t-white rounded-full" style={{ animation: 'spin 0.7s linear infinite' }} /> Génération...</> : <><i className="fas fa-magic"></i> Générer avec l&apos;IA</>}
   199→            </button>
   200→            <div className="text-center text-[0.65rem] text-[#94A3B8] mb-2">— ou —</div>
   201→            <label className="w-full py-3 rounded-xl border-2 border-dashed border-[rgba(0,0,0,0.1)] text-[#64748B] font-semibold text-[0.78rem] cursor-pointer flex items-center justify-center gap-2 hover:border-[#22C55E] hover:text-[#22C55E] transition-colors">
   202→              <i className="fas fa-upload"></i> Uploader
   203→              <input type="file" accept="image/*" className="hidden" onChange={e => e.target.files?.[0] && handleUpload(e.target.files[0])} />
   204→            </label>
   205→            <div className="mt-3 text-center text-[0.6rem] text-[#94A3B8]">+{selCamp.rewardCfa} FCFA/image validée • Max {selCamp.dailyLimit}/jour</div>
   206→          </div>
   207→        </div>
   208→      )}
   209→    </>
   210→  );
   211→}

   288→function CampV({ camps, onSel }: { camps: Campaign[]; onSel: (c: Campaign) => void }) {
   289→  if (!camps.length) return <div className="text-center py-12"><div className="w-16 h-16 rounded-full bg-[rgba(0,0,0,0.04)] flex items-center justify-center mx-auto mb-3"><i className="fas fa-bullhorn text-[#94A3B8] text-xl"></i></div><div className="text-[0.85rem] font-semibold text-[#64748B]">Aucune campagne active</div><div className="text-[0.72rem] text-[#94A3B8]">De nouvelles missions arriveront bientôt !</div></div>;
   290→  return (
   291→    <div className="space-y-3">
   292→      {camps.map(c => (
   293→        <div key={c.id} className="bg-white rounded-2xl p-4 shadow-[0_1px_3px_rgba(0,0,0,0.04)] border border-[rgba(0,0,0,0.03)]">
   294→          <div className="flex items-start gap-3 mb-3">
   295→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-[#22C55E] to-[#16A34A] flex items-center justify-center shrink-0"><i className="fas fa-image text-white text-[0.85rem]"></i></div>
   296→            <div className="flex-1 min-w-0"><div className="text-[0.85rem] font-bold text-[#1F2937]">{c.brand}</div><div className="text-[0.68rem] text-[#64748B]">{c.name}</div></div>
   297→            <span className="px-2 py-0.5 rounded-full text-[0.55rem] font-bold bg-[rgba(34,197,94,0.1)] text-[#22C55E]">+{c.rewardCfa}F</span>
   298→          </div>
   299→          <div className="text-[0.68rem] text-[#64748B] mb-3 leading-relaxed line-clamp-2">{c.brief}</div>
   300→          <div className="flex items-center gap-2 mb-3 text-[0.6rem] text-[#94A3B8]">
