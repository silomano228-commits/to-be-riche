   511→      projectBalance: 'Projet',
   512→      investBalance: 'Investissement',

   519→    // Detect changed fields and validate every input (must parse to >=0 numbers).

   522→    for (const k of Object.keys(editBalanceDraft) as FieldKey[]) {
   523→      const raw = editBalanceDraft[k];
