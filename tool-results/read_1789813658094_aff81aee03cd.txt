    59→export default function MissionsScreen() {
    60→  const { user, setPage, addToast } = useAppStore();
    61→  const [tab, setTab] = useState<'dashboard'|'campaigns'|'myimages'|'eligibility'|'loans'>('dashboard');
    62→  const [dash, setDash] = useState<DashboardData|null>(null);
    63→  const [campaigns, setCampaigns] = useState<Campaign[]>([]);
    64→  const [images, setImages] = useState<MissionImage[]>([]);
    65→  const [loading, setLoading] = useState(true);
    66→  const [uploading, setUploading] = useState(false);
    67→  const [selCamp, setSelCamp] = useState<Campaign|null>(null);
    68→  const [showUpload, setShowUpload] = useState(false);
    69→  const [elig, setElig] = useState<any>(null);
    70→  const [loans, setLoans] = useState<any[]>([]);
    71→  const fileRef = useRef<HTMLInputElement>(null);
    72→
    73→  const loadData = async () => {
    74→    setLoading(true);
    75→    try {
    76→      const [dr, cr, ir] = await Promise.all([
    77→        authFetch('/api/missions/dashboard'),
    78→        authFetch('/api/missions/campaigns'),
    79→        authFetch('/api/missions/images?limit=20'),
    80→      ]);
    81→      const dd = await dr.json(); if (dd.success) setDash(dd.dashboard);
    82→      const cd = await cr.json(); if (cd.success) setCampaigns(cd.campaigns);
    83→      const id2 = await ir.json(); if (id2.success) setImages(id2.images);
    84→    } catch {}
    85→    setLoading(false);
    86→  };
    87→
    88→  useEffect(() => { loadData(); // eslint-disable-line react-hooks/set-state-in-effect }, []);
    89→
    90→  const loadElig = async () => {
    91→    try { const r = await authFetch('/api/missions/eligibility'); const d = await r.json(); if (d.success) setElig({ ...d.eligibility, userLevel: d.userLevel, userLevelLabel: d.userLevelLabel, activeLoan: d.activeLoan }); } catch {}
    92→  };
    93→
    94→  const loadLoans = async () => {
    95→    try { const r = await authFetch('/api/missions/loans'); const d = await r.json(); if (d.success) setLoans(d.loans); } catch {}
    96→  };
    97→
    98→  const handleUpload = async (file: File) => {
    99→    if (!selCamp) return;
   100→    setUploading(true);
   101→    const reader = new FileReader();
   102→    reader.onload = async (e) => {
   103→      const b64 = (e.target?.result as string).split(',')[1];
   104→      try {
   105→        const r = await authFetch('/api/missions/images', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ campaignId: selCamp.id, imageData: b64 }) });
   106→        const d = await r.json();
   107→        if (d.success) { addToast('Image envoyée ! Validation IA en cours...', 'success'); setShowUpload(false); setSelCamp(null); setTimeout(loadData, 3000); }
   108→        else addToast(d.error || 'Erreur', 'error');
   109→      } catch { addToast('Erreur d\'envoi', 'error'); }
   110→      setUploading(false);
   111→    };
   112→    reader.readAsDataURL(file);
   113→  };
   114→
   115→  const reqLoan = async (amt: number) => {
   116→    try {
   117→      const r = await authFetch('/api/missions/loans', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ amountCfa: amt }) });
   118→      const d = await r.json();
   119→      if (d.success) { addToast(`Demande de ${formatCfa(amt)} enregistrée !`, 'success'); loadLoans(); loadElig(); }
   120→      else addToast(d.error || 'Conditions non remplies', 'error');
   121→    } catch { addToast('Erreur', 'error'); }
   122→  };
   123→
   124→  const depositCaution = async () => {
   125→    try {
   126→      const r = await authFetch('/api/missions/caution', { method: 'POST' });
   127→      const d = await r.json();
   128→      if (d.success) { addToast('Caution constituée !', 'success'); loadData(); }
   129→      else addToast(d.error || 'Erreur', 'error');
   130→    } catch { addToast('Erreur', 'error'); }
   131→  };
   132→
   133→  const repay = async (lid: string) => {
   134→    const s = prompt('Montant à rembourser (FCFA) :');
   135→    if (!s) return;
   136→    const amt = parseInt(s);
   137→    if (isNaN(amt) || amt <= 0) { addToast('Montant invalide', 'error'); return; }
   138→    try {
   139→      const r = await authFetch(`/api/missions/loans/${lid}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'repay', amountCfa: amt }) });
   140→      const d = await r.json();
   141→      if (d.success) { addToast('Remboursement effectué !', 'success'); loadLoans(); loadData(); refreshUser(); }
   142→      else addToast(d.error || 'Erreur', 'error');
   143→    } catch { addToast('Erreur', 'error'); }
   144→  };
   145→
   146→  if (!user) return null;
   147→
   148→  const tabs = [
   149→    { id: 'dashboard' as const, icon: 'fa-th-large', label: 'Tableau' },
   150→    { id: 'campaigns' as const, icon: 'fa-bullhorn', label: 'Missions' },
   151→    { id: 'myimages' as const, icon: 'fa-images', label: 'Mes images' },
   152→    { id: 'eligibility' as const, icon: 'fa-chart-bar', label: 'Éligibilité' },
   153→    { id: 'loans' as const, icon: 'fa-hand-holding-usd', label: 'Prêts' },
   154→  ];
   155→
   156→  return (
   157→    <>
   158→      <Header title="Espace Jeunes" rightElement={
   159→        <button onClick={() => setPage('profile')} className="w-9 h-9 rounded-[10px] flex items-center justify-center bg-[rgba(0,0,0,0.04)] text-[#64748B] cursor-pointer border-none text-[0.85rem]"><i className="far fa-user-circle"></i></button>
   160→      } />
   161→      <div className="flex gap-1 px-4 pt-2 pb-1 overflow-x-auto bg-white border-b border-[rgba(0,0,0,0.05)]">
   162→        {tabs.map(t => (
   163→          <button key={t.id} onClick={() => { setTab(t.id); if (t.id === 'eligibility') loadElig(); if (t.id === 'loans') loadLoans(); }}
   164→            className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-[0.72rem] font-semibold border-none cursor-pointer transition-all whitespace-nowrap ${tab === t.id ? 'bg-[#22C55E] text-white' : 'bg-[rgba(0,0,0,0.03)] text-[#64748B]'}`}>
   165→            <i className={`fas ${t.icon} text-[0.65rem]`}></i> {t.label}
   166→          </button>
   167→        ))}
   168→      </div>
   169→      <div className="flex-1 overflow-y-auto px-4 py-4">
   170→        {loading && !dash ? <div className="flex items-center justify-center py-12"><div className="w-6 h-6 border-[2.5px] border-[rgba(0,0,0,0.08)] border-t-[#22C55E] rounded-full" style={{ animation: 'spin 0.7s linear infinite' }} /></div>
   171→        : tab === 'dashboard' ? <DashV dash={dash} go={setTab} />
   172→        : tab === 'campaigns' ? <CampV camps={campaigns} onSel={c => { setSelCamp(c); setShowUpload(true); }} />
   173→        : tab === 'myimages' ? <ImgV imgs={images} />
   174→        : tab === 'eligibility' ? <EligV elig={elig} onLoan={reqLoan} onCaution={depositCaution} dash={dash} onLoad={loadElig} />
   175→        : tab === 'loans' ? <LoanV loans={loans} onRepay={repay} />
   176→        : null}
   177→      </div>
   178→
   179→      {/* Upload modal — primary way to submit images */}
   180→      {showUpload && selCamp && (
   181→        <div className="fixed inset-0 z-50 bg-[rgba(0,0,0,0.6)] flex items-end justify-center" onClick={() => setShowUpload(false)}>
   182→          <div className="bg-white w-full max-w-lg rounded-t-2xl p-5 max-h-[85vh] overflow-y-auto" onClick={e => e.stopPropagation()}>
   183→            <div className="flex justify-between items-center mb-4">
   184→              <h3 className="text-lg font-bold text-[#1F2937]">Poster une image</h3>
   185→              <button onClick={() => setShowUpload(false)} className="text-[#94A3B8] text-xl cursor-pointer border-none bg-transparent"><i className="fas fa-times"></i></button>
   186→            </div>
   187→
   188→            {/* Campaign info */}
   189→            <div className="mb-4 p-3.5 bg-[rgba(34,197,94,0.06)] rounded-xl border border-[rgba(34,197,94,0.12)]">
   190→              <div className="flex items-center gap-2 mb-2">
   191→                <div className="w-8 h-8 rounded-lg flex items-center justify-center" style={{ background: (CAT_ICON[selCamp.category] || CAT_ICON.general).color + '15' }}>
   192→                  <i className={`fas ${(CAT_ICON[selCamp.category] || CAT_ICON.general).icon} text-[0.75rem]`} style={{ color: (CAT_ICON[selCamp.category] || CAT_ICON.general).color }}></i>
   193→                </div>
   194→                <div>
   195→                  <div className="text-[0.78rem] font-bold text-[#1F2937]">{selCamp.brand}</div>
   196→                  <div className="text-[0.65rem] text-[#64748B]">{selCamp.name}</div>
   197→                </div>
   198→              </div>
   199→              <div className="text-[0.65rem] text-[#64748B] leading-relaxed">{selCamp.brief}</div>
   200→              {selCamp.constraints && <div className="text-[0.6rem] text-[#94A3B8] mt-1.5 italic">📌 {selCamp.constraints}</div>}
   201→            </div>
   202→
   203→            {/* Instructions - CRITICAL: explain external creation */}
   204→            <div className="mb-4 p-3.5 bg-[rgba(59,130,246,0.06)] rounded-xl border border-[rgba(59,130,246,0.12)]">
   205→              <div className="text-[0.72rem] font-bold text-[#3B82F6] mb-2"><i className="fas fa-info-circle mr-1"></i> Comment ça marche ?</div>
   206→              <ol className="text-[0.65rem] text-[#1E40AF] space-y-1.5 list-decimal ml-3.5">
   207→                <li>Ouvrez <strong>ChatGPT, DALL-E, Midjourney</strong> ou toute autre IA</li>
   208→                <li>Générez une image en suivant le brief ci-dessus</li>
   209→                <li>Téléchargez l&apos;image sur votre appareil</li>
   210→                <li>Uploadez-la ici en cliquant le bouton ci-dessous</li>
   211→              </ol>
   212→            </div>
   213→
   214→            {/* Upload button */}
   215→            <label className="w-full py-4 rounded-xl bg-gradient-to-r from-[#22C55E] to-[#16A34A] text-white font-semibold text-[0.88rem] cursor-pointer flex items-center justify-center gap-2 shadow-[0_4px_20px_rgba(34,197,94,0.25)] transition-transform active:scale-[0.97] mb-3">
   216→              {uploading ? (
   217→                <><div className="w-4 h-4 border-[2px] border-white/30 border-t-white rounded-full" style={{ animation: 'spin 0.7s linear infinite' }} /> Envoi en cours...</>
   218→              ) : (
   219→                <><i className="fas fa-cloud-upload-alt text-[1rem]"></i> Uploader mon image</>
   220→              )}
   221→              <input type="file" accept="image/*" className="hidden" onChange={e => e.target.files?.[0] && handleUpload(e.target.files[0])} disabled={uploading} />
   222→            </label>
   223→
   224→            {/* Reward info */}
   225→            <div className="flex items-center justify-center gap-4 text-[0.65rem] text-[#94A3B8]">
   226→              <span><i className="fas fa-coins text-[#F59E0B] mr-1"></i> +{selCamp.rewardCfa} FCFA/image validée</span>
   227→              <span><i className="fas fa-clock mr-1"></i> Max {selCamp.dailyLimit}/jour</span>
   228→              <span><i className="fas fa-expand mr-1"></i> Format {selCamp.format}</span>
   229→            </div>
   230→          </div>
   231→        </div>
   232→      )}
   233→    </>
   234→  );
   235→}

   237→/* ============ DASHBOARD VIEW ============ */
   238→function DashV({ dash, go }: { dash: DashboardData|null; go: (p: any) => void }) {
   239→  if (!dash) return <div className="text-center text-[#94A3B8] py-8">Chargement...</div>;
   240→  const pct = Math.min(100, (dash.missionTotalEarnedCfa / dash.objectiveCfa) * 100);
   241→  return (
   242→    <div>
   243→      {/* Balance card */}
   244→      <div className="bg-gradient-to-br from-[#0F172A] via-[#1E293B] to-[#0F172A] text-white rounded-2xl p-5 mb-4 relative overflow-hidden">
   245→        <div className="absolute -top-12 -right-12 w-[140px] h-[140px] bg-[radial-gradient(circle,rgba(34,197,94,0.15),transparent_65%)]" />
   246→        <div className="relative z-[1]">
   247→          <div className="text-[0.65rem] text-[rgba(255,255,255,0.5)] mb-1">Solde missions</div>
   248→          <div className="text-[1.8rem] font-black tracking-[-1px]">{formatCfa(dash.missionBalanceCfa)}</div>
   249→          <div className="flex gap-3 mt-3">
   250→            <div className="flex-1 bg-[rgba(255,255,255,0.08)] rounded-lg p-2.5">
   251→              <div className="text-[0.55rem] text-[rgba(255,255,255,0.4)]">Gains du jour</div>
   252→              <div className="text-[0.85rem] font-bold text-[#22C55E]">+{dash.validatedToday * dash.rewardPerImage} FCFA</div>
   253→            </div>
   254→            <div className="flex-1 bg-[rgba(255,255,255,0.08)] rounded-lg p-2.5">
   255→              <div className="text-[0.55rem] text-[rgba(255,255,255,0.4)]">Validées aujourd&apos;hui</div>
   256→              <div className="text-[0.85rem] font-bold">{dash.validatedToday}/{dash.dailyLimit}</div>
   257→            </div>
   258→          </div>
   259→        </div>
   260→      </div>
   261→
   262→      {/* Progress toward loan eligibility */}
   263→      <div className="bg-white rounded-2xl p-4 mb-3 shadow-[0_1px_3px_rgba(0,0,0,0.04)] border border-[rgba(0,0,0,0.03)]">
   264→        <div className="flex justify-between items-center mb-2">
   265→          <div className="text-[0.78rem] font-bold text-[#1F2937]">Objectif pour les prêts</div>
   266→          <div className="text-[0.72rem] font-semibold text-[#22C55E]">{dash.missionTotalEarnedCfa}/{dash.objectiveCfa} FCFA</div>
   267→        </div>
   268→        <div className="w-full h-2.5 bg-[rgba(0,0,0,0.05)] rounded-full overflow-hidden">
   269→          <div className="h-full bg-gradient-to-r from-[#22C55E] to-[#16A34A] rounded-full transition-all duration-500" style={{ width: `${pct}%` }} />
   270→        </div>
   271→        <div className="text-[0.6rem] text-[#94A3B8] mt-1">{pct >= 100 ? '🎉 Objectif atteint ! Vous pouvez demander un prêt.' : `${Math.round(pct)}% — il vous manque ${dash.objectiveCfa - dash.missionTotalEarnedCfa} FCFA`}</div>
   272→      </div>
   273→
   274→      {/* Quick stats */}
   275→      <div className="grid grid-cols-2 gap-2.5 mb-3">
   276→        {[
   277→          { icon: 'fa-bullhorn', ic: '#22C55E', ib: 'rgba(34,197,94,0.1)', l: 'Campagnes actives', v: String(dash.activeCampaigns), go: () => go('campaigns') },
   278→          { icon: 'fa-shield-alt', ic: '#3B82F6', ib: 'rgba(59,130,246,0.1)', l: 'Caution', v: dash.cautionStatus === 'active' ? 'Constituée' : 'Non constituée', go: () => go('eligibility') },
   279→          { icon: 'fa-users', ic: '#A855F7', ib: 'rgba(168,85,247,0.1)', l: 'Parrainages', v: `${dash.validatedReferralCount}/5`, go: undefined },
   280→          { icon: 'fa-hand-holding-usd', ic: '#F59E0B', ib: 'rgba(245,158,11,0.1)', l: 'Prêts', v: dash.activeLoans > 0 ? `${dash.activeLoans} actif(s)` : 'Aucun', go: () => go('loans') },
   281→        ].map((c, i) => (
   282→          <div key={i} className="bg-white rounded-xl p-3.5 shadow-[0_1px_3px_rgba(0,0,0,0.04)] border border-[rgba(0,0,0,0.03)]" onClick={c.go} style={c.go ? { cursor: 'pointer' } : {}}>
   283→            <div className="w-8 h-8 rounded-lg flex items-center justify-center mb-2" style={{ background: c.ib }}><i className={`fas ${c.icon} text-[0.7rem]`} style={{ color: c.ic }}></i></div>
   284→            <div className="text-[0.65rem] text-[#94A3B8]">{c.l}</div>
   285→            <div className="text-[0.85rem] font-bold text-[#1F2937]">{c.v}</div>
   286→          </div>
   287→        ))}
   288→      </div>
   289→
   290→      {/* How it works */}
   291→      <div className="bg-white rounded-2xl p-4 shadow-[0_1px_3px_rgba(0,0,0,0.04)] border border-[rgba(0,0,0,0.03)] mb-3">
   292→        <div className="text-[0.78rem] font-bold text-[#1F2937] mb-3">Comment gagner de l&apos;argent ?</div>
   293→        {[
   294→          { s: 1, t: 'Choisissez une campagne de mission', c: '#22C55E', icon: 'fa-bullhorn' },
   295→          { s: 2, t: 'Générez une image avec ChatGPT, DALL-E, Midjourney...', c: '#3B82F6', icon: 'fa-magic' },
   296→          { s: 3, t: 'Uploadez votre image ici (+25 FCFA si validée)', c: '#F59E0B', icon: 'fa-upload' },
   297→          { s: 4, t: 'Atteignez 2 500 FCFA pour débloquer les prêts', c: '#A855F7', icon: 'fa-unlock' },
   298→          { s: 5, t: 'Constituez la caution + parrainages → Demandez votre prêt !', c: '#EF4444', icon: 'fa-hand-holding-usd' },
   299→        ].map((x, i) => (
   300→          <div key={i} className={`flex items-start gap-2.5 ${i < 4 ? 'mb-2.5' : ''}`}>
   301→            <div className="w-7 h-7 rounded-full flex items-center justify-center text-white font-bold text-[0.6rem] shrink-0" style={{ background: x.c }}>{x.s}</div>
   302→            <div className="text-[0.72rem] text-[#64748B] pt-1 flex-1">{x.t}</div>
   303→            <i className={`fas ${x.icon} text-[0.7rem] mt-1`} style={{ color: x.c }}></i>
   304→          </div>
   305→        ))}
   306→      </div>
   307→
   308→      {/* Important rules */}
   309→      <div className="bg-[rgba(245,158,11,0.06)] rounded-xl p-3 border border-[rgba(245,158,11,0.1)]">
   310→        <div className="text-[0.72rem] font-bold text-[#F59E0B] mb-1.5"><i className="fas fa-exclamation-triangle mr-1"></i> Règles importantes</div>
   311→        <ul className="text-[0.62rem] text-[#92400E] space-y-1">
   312→          <li>• Créez vos images avec une IA externe (ChatGPT, DALL-E, Midjourney...)</li>
   313→          <li>• Uploadez-les ici — l&apos;IA valide automatiquement la conformité</li>
   314→          <li>• Maximum 10 images rémunérées par jour</li>
   315→          <li>• Les images trop similaires sont refusées (doublons)</li>
   316→          <li>• Les images disparaissent le lendemain pour libérer l&apos;espace</li>
   317→          <li>• La caution n&apos;est pas disponible pour les retraits</li>
   318→        </ul>
   319→      </div>
   320→    </div>
   321→  );
