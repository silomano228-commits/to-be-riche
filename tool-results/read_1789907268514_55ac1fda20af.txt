    59→export default function MissionsScreen() {
    60→  const { user, setPage, addToast } = useAppStore();

    62→  const [dash, setDash] = useState<DashboardData|null>(null);
    63→  const [campaigns, setCampaigns] = useState<Campaign[]>([]);
    64→  const [images, setImages] = useState<MissionImage[]>([]);
    65→  const [loading, setLoading] = useState(true);
    66→  const [uploading, setUploading] = useState(false);
    67→  const [selCamp, setSelCamp] = useState<Campaign|null>(null);
    68→  const [showUpload, setShowUpload] = useState(false);
    69→  const [elig, setElig] = useState<any>(null);
    70→  const [loans, setLoans] = useState<any[]>([]);
    71→  const fileRef = useRef<HTMLInputElement>(null);
    72→
    73→  const loadData = async () => {
    74→    setLoading(true);
    75→    try {
    76→      const [dr, cr, ir] = await Promise.all([
    77→        authFetch('/api/missions/dashboard'),
    78→        authFetch('/api/missions/campaigns'),
    79→        authFetch('/api/missions/images?limit=20'),
    80→      ]);
    81→      const dd = await dr.json(); if (dd.success) setDash(dd.dashboard);
    82→      const cd = await cr.json(); if (cd.success) setCampaigns(cd.campaigns);
    83→      const id2 = await ir.json(); if (id2.success) setImages(id2.images);
    84→    } catch {}
    85→    setLoading(false);
    86→  };
    87→
    88→  useEffect(() => { loadData(); // eslint-disable-line react-hooks/set-state-in-effect
    89→  }, []);
    90→
    91→  const loadElig = async () => {
    92→    try { const r = await authFetch('/api/missions/eligibility'); const d = await r.json(); if (d.success) setElig({ ...d.eligibility, userLevel: d.userLevel, userLevelLabel: d.userLevelLabel, activeLoan: d.activeLoan }); } catch {}
    93→  };
    94→
    95→  const loadLoans = async () => {
    96→    try { const r = await authFetch('/api/missions/loans'); const d = await r.json(); if (d.success) setLoans(d.loans); } catch {}
    97→  };
    98→
    99→  const handleUpload = async (file: File) => {
   100→    if (!selCamp) return;
   101→    setUploading(true);
   102→    const reader = new FileReader();
   103→    reader.onload = async (e) => {
   104→      const b64 = (e.target?.result as string).split(',')[1];
   105→      try {
   106→        const r = await authFetch('/api/missions/images', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ campaignId: selCamp.id, imageData: b64 }) });
   107→        const d = await r.json();
   108→        if (d.success) { addToast('Image importée ! Analyse IA en cours...', 'success'); setShowUpload(false); setSelCamp(null); setTimeout(loadData, 3000); }
   109→        else addToast(d.error || 'Erreur', 'error');
   110→      } catch { addToast('Erreur d\'envoi', 'error'); }
   111→      setUploading(false);
   112→    };
   113→    reader.readAsDataURL(file);
   114→  };
   115→
   116→  const reqLoan = async (amt: number) => {
   117→    try {
   118→      const r = await authFetch('/api/missions/loans', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ amountCfa: amt }) });
   119→      const d = await r.json();
   120→      if (d.success) { addToast(`Demande de ${formatCfa(amt)} enregistrée !`, 'success'); loadLoans(); loadElig(); }
   121→      else addToast(d.error || 'Conditions non remplies', 'error');
   122→    } catch { addToast('Erreur', 'error'); }
   123→  };
   124→
   125→  const depositCaution = async () => {
   126→    try {
   127→      const r = await authFetch('/api/missions/caution', { method: 'POST' });
   128→      const d = await r.json();
   129→      if (d.success) { addToast('Caution constituée !', 'success'); loadData(); }
   130→      else addToast(d.error || 'Erreur', 'error');
   131→    } catch { addToast('Erreur', 'error'); }
   132→  };
   133→
   134→  const repay = async (lid: string) => {
   135→    const s = prompt('Montant à rembourser (FCFA) :');
   136→    if (!s) return;
   137→    const amt = parseInt(s);
   138→    if (isNaN(amt) || amt <= 0) { addToast('Montant invalide', 'error'); return; }
   139→    try {
   140→      const r = await authFetch(`/api/missions/loans/${lid}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'repay', amountCfa: amt }) });
   141→      const d = await r.json();
   142→      if (d.success) { addToast('Remboursement effectué !', 'success'); loadLoans(); loadData(); refreshUser(); }
   143→      else addToast(d.error || 'Erreur', 'error');
   144→    } catch { addToast('Erreur', 'error'); }
   145→  };
   146→
   147→  if (!user) return null;
   148→
   149→  const tabs = [
   150→    { id: 'dashboard' as const, icon: 'fa-th-large', label: 'Tableau' },
   151→    { id: 'campaigns' as const, icon: 'fa-bullhorn', label: 'Tâches' },
   152→    { id: 'myimages' as const, icon: 'fa-images', label: 'Mes images' },
   153→    { id: 'eligibility' as const, icon: 'fa-chart-bar', label: 'Éligibilité' },
   154→    { id: 'loans' as const, icon: 'fa-hand-holding-usd', label: 'Prêts' },
   155→  ];
   156→
   157→  return (
   158→    <>
   159→      <Header title="Jeune Élan" rightElement={
   160→        <button onClick={() => setPage('profile')} className="w-9 h-9 rounded-[10px] flex items-center justify-center bg-[rgba(0,0,0,0.04)] text-[#64748B] cursor-pointer border-none text-[0.85rem]"><i className="far fa-user-circle"></i></button>
   161→      } />
   162→      <div className="flex gap-1 px-4 pt-2 pb-1 overflow-x-auto bg-white border-b border-[rgba(0,0,0,0.05)]">
   163→        {tabs.map(t => (
   164→          <button key={t.id} onClick={() => { setTab(t.id); if (t.id === 'eligibility') loadElig(); if (t.id === 'loans') loadLoans(); }}
   165→            className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-[0.72rem] font-semibold border-none cursor-pointer transition-all whitespace-nowrap ${tab === t.id ? 'bg-[#22C55E] text-white' : 'bg-[rgba(0,0,0,0.03)] text-[#64748B]'}`}>
   166→            <i className={`fas ${t.icon} text-[0.65rem]`}></i> {t.label}
   167→          </button>
   168→        ))}
   169→      </div>
   170→      <div className="flex-1 overflow-y-auto px-4 py-4">
   171→        {loading && !dash ? <div className="flex items-center justify-center py-12"><div className="w-6 h-6 border-[2.5px] border-[rgba(0,0,0,0.08)] border-t-[#22C55E] rounded-full" style={{ animation: 'spin 0.7s linear infinite' }} /></div>
   172→        : tab === 'dashboard' ? <DashV dash={dash} go={setTab} />
   173→        : tab === 'campaigns' ? <CampV camps={campaigns} onSel={c => { setSelCamp(c); setShowUpload(true); }} />
   174→        : tab === 'myimages' ? <ImgV imgs={images} />
   175→        : tab === 'eligibility' ? <EligV elig={elig} onLoan={reqLoan} onCaution={depositCaution} dash={dash} onLoad={loadElig} />
   176→        : tab === 'loans' ? <LoanV loans={loans} onRepay={repay} />
   177→        : null}
   178→      </div>
   179→
   180→      {/* Upload modal — primary way to submit images */}
   181→      {showUpload && selCamp && (
   182→        <div className="fixed inset-0 z-50 bg-[rgba(0,0,0,0.6)] flex items-end justify-center" onClick={() => setShowUpload(false)}>
   183→          <div className="bg-white w-full max-w-lg rounded-t-2xl p-5 max-h-[85vh] overflow-y-auto" onClick={e => e.stopPropagation()}>
   184→            <div className="flex justify-between items-center mb-4">
   185→              <h3 className="text-lg font-bold text-[#1F2937]">Poster une image</h3>
   186→              <button onClick={() => setShowUpload(false)} className="text-[#94A3B8] text-xl cursor-pointer border-none bg-transparent"><i className="fas fa-times"></i></button>
   187→            </div>
   188→
   189→            {/* Campaign info */}
   190→            <div className="mb-4 p-3.5 bg-[rgba(34,197,94,0.06)] rounded-xl border border-[rgba(34,197,94,0.12)]">
   191→              <div className="flex items-center gap-2 mb-2">
   192→                <div className="w-8 h-8 rounded-lg flex items-center justify-center" style={{ background: (CAT_ICON[selCamp.category] || CAT_ICON.general).color + '15' }}>
   193→                  <i className={`fas ${(CAT_ICON[selCamp.category] || CAT_ICON.general).icon} text-[0.75rem]`} style={{ color: (CAT_ICON[selCamp.category] || CAT_ICON.general).color }}></i>
   194→                </div>
   195→                <div>
   196→                  <div className="text-[0.78rem] font-bold text-[#1F2937]">{selCamp.brand}</div>
   197→                  <div className="text-[0.65rem] text-[#64748B]">{selCamp.name}</div>
   198→                </div>
   199→              </div>
   200→              <div className="text-[0.65rem] text-[#64748B] leading-relaxed">{selCamp.brief}</div>
   201→              {selCamp.constraints && <div className="text-[0.6rem] text-[#94A3B8] mt-1.5 italic">📌 {selCamp.constraints}</div>}
   202→            </div>
   203→
   204→            {/* Instructions - CRITICAL: explain external creation */}
   205→            <div className="mb-4 p-3.5 bg-[rgba(59,130,246,0.06)] rounded-xl border border-[rgba(59,130,246,0.12)]">
   206→              <div className="text-[0.72rem] font-bold text-[#3B82F6] mb-2"><i className="fas fa-info-circle mr-1"></i> Comment ça marche ?</div>
   207→              <ol className="text-[0.65rem] text-[#1E40AF] space-y-1.5 list-decimal ml-3.5">
   208→                <li>Ouvrez <strong>ChatGPT, Gemini, Midjourney</strong> ou toute autre IA</li>
   209→                <li>Créez une image en suivant le brief ci-dessus</li>
   210→                <li>Téléchargez l&apos;image sur votre appareil</li>
   211→                <li>Importez-la ici en cliquant le bouton ci-dessous</li>
   212→              </ol>
   213→            </div>
   214→
   215→            {/* Upload button */}
   216→            <label className="w-full py-4 rounded-xl bg-gradient-to-r from-[#22C55E] to-[#16A34A] text-white font-semibold text-[0.88rem] cursor-pointer flex items-center justify-center gap-2 shadow-[0_4px_20px_rgba(34,197,94,0.25)] transition-transform active:scale-[0.97] mb-3">
   217→              {uploading ? (
   218→                <><div className="w-4 h-4 border-[2px] border-white/30 border-t-white rounded-full" style={{ animation: 'spin 0.7s linear infinite' }} /> Analyse en cours...</>

   220→                <><i className="fas fa-cloud-upload-alt text-[1rem]"></i> Importer mon image</>

   225→            {/* Reward info */}

   229→              <span><i className="fas fa-expand mr-1"></i> Format {selCamp.format}</span>

   231→          </div>
   232→        </div>

   234→    </>

   239→const LEVEL_LABELS: Record<number, { label: string; color: string }> = {
   240→  1: { label: 'Nouveau', color: '#94A3B8' },
   241→  2: { label: 'Actif', color: '#F59E0B' },
   242→  3: { label: 'Éligible', color: '#22C55E' },

   245→

   249→  const lvl = LEVEL_LABELS[dash.userLevel] || LEVEL_LABELS[1];
   250→  return (
   251→    <div>
   252→      {/* Balance card */}
   253→      <div className="bg-gradient-to-br from-[#0F172A] via-[#1E293B] to-[#0F172A] text-white rounded-2xl p-5 mb-4 relative overflow-hidden">

   255→        <div className="relative z-[1]">
   256→          <div className="text-[0.65rem] text-[rgba(255,255,255,0.5)] mb-1">Solde tâches</div>

   280→          <div className="text-[0.78rem] font-bold text-[#1F2937]">Objectif pour les prêts</div>
   281→          <div className="text-[0.72rem] font-semibold text-[#22C55E]">{dash.missionTotalEarnedCfa}/{dash.objectiveCfa} FCFA</div>
   282→        </div>
   283→        <div className="w-full h-2.5 bg-[rgba(0,0,0,0.05)] rounded-full overflow-hidden">

   288→
   289→      {/* Quick stats */}
   290→      <div className="grid grid-cols-2 gap-2.5 mb-3">
   291→        {[
   292→          { icon: 'fa-bullhorn', ic: '#22C55E', ib: 'rgba(34,197,94,0.1)', l: 'Campagnes actives', v: String(dash.activeCampaigns), go: () => go('campaigns') },
   293→          { icon: 'fa-shield-alt', ic: '#3B82F6', ib: 'rgba(59,130,246,0.1)', l: 'Caution', v: dash.cautionStatus === 'active' ? 'Constituée' : 'Non constituée', go: () => go('eligibility') },
   294→          { icon: 'fa-users', ic: '#A855F7', ib: 'rgba(168,85,247,0.1)', l: 'Parrainages', v: `${dash.validatedReferralCount}/5`, go: undefined },

   299→            <div className="text-[0.65rem] text-[#94A3B8]">{c.l}</div>

   306→      <div className="bg-white rounded-2xl p-4 shadow-[0_1px_3px_rgba(0,0,0,0.04)] border border-[rgba(0,0,0,0.03)] mb-3">
   307→        <div className="text-[0.78rem] font-bold text-[#1F2937] mb-3">Comment gagner de l&apos;argent ?</div>
   308→        {[
   309→          { s: 1, t: 'Choisissez une campagne de tâche', c: '#22C55E', icon: 'fa-bullhorn' },
   310→          { s: 2, t: 'Créez une image avec ChatGPT, Gemini, Midjourney...', c: '#3B82F6', icon: 'fa-magic' },
   311→          { s: 3, t: 'Importez votre image ici (+25 FCFA si validée)', c: '#F59E0B', icon: 'fa-upload' },
   312→          { s: 4, t: 'Atteignez 2 500 FCFA pour débloquer les prêts', c: '#A855F7', icon: 'fa-unlock' },
   313→          { s: 5, t: 'Constituez la caution + parrainages → Demandez votre prêt !', c: '#EF4444', icon: 'fa-hand-holding-usd' },
   314→        ].map((x, i) => (
