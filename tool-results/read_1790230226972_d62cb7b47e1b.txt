     9→// ==================== ERROR BOUNDARY ====================
    10→class ErrorBoundary extends Component<{ children: React.ReactNode }, { hasError: boolean; error?: string }> {
    11→  state = { hasError: false, error: '' };
    12→  static getDerivedStateFromError(e: Error) { return { hasError: true, error: e.message }; }
    13→  render() {
    14→    if (this.state.hasError) {
    15→      return (
    16→        <div className="bg-gradient-to-b from-[#F8F9FA] to-[#F1F5F9] min-h-screen flex items-center justify-center p-6">
    17→          <div className="text-center max-w-[320px]">
    18→            <div className="w-16 h-16 rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mx-auto mb-4">
    19→              <i className="fas fa-exclamation-triangle text-[#F59E0B] text-2xl"></i>
    20→            </div>
    21→            <h2 className="text-lg font-bold text-[#1F2937] mb-2">Oups !</h2>
    22→            <p className="text-sm text-[rgba(0,0,0,0.5)] mb-4">Une erreur inattendue s&apos;est produite.</p>
    23→            <button onClick={() => { this.setState({ hasError: false }); window.location.reload(); }} className="px-6 py-3 rounded-xl bg-[#22C55E] text-white font-semibold border-none cursor-pointer">
    24→              Réessayer
    25→            </button>
    26→          </div>
    27→        </div>
    28→      );
    29→    }
    30→    return this.props.children;
    31→  }
    32→}

    83→// ==================== AUTH SCREEN ====================
    84→function AuthScreen() {
    85→  const { user, setUser, setPage, addToast } = useAppStore();
    86→  // Referral code from URL (lazy init before useState)
    87→  const { prefilledReferral, initialMode } = (() => {
    88→    if (typeof window === 'undefined') return { prefilledReferral: '', initialMode: 'login' as const };
    89→    try {
    90→      const params = new URLSearchParams(window.location.search);
    91→      const ref = params.get('ref');
    92→      if (ref) return { prefilledReferral: ref.toUpperCase(), initialMode: 'register' as const };
    93→    } catch { /* */ }
    94→    return { prefilledReferral: '', initialMode: 'login' as const };
    95→  })();
    96→  const [mode, setMode] = useState<'login' | 'register'>(initialMode);
    97→  const [loading, setLoading] = useState(false);
    98→  const [showPw, setShowPw] = useState({ l: false, r: false, r2: false });
    99→  const [errors, setErrors] = useState<Record<string, string>>({});
   100→  // OTP state (registration only)
   101→  const [otpStep, setOtpStep] = useState(false);
   102→  const [otpEmail, setOtpEmail] = useState('');
   103→  const [otpCode, setOtpCode] = useState('');
   104→  const [otpLoading, setOtpLoading] = useState(false);
   105→  const [simCode, setSimCode] = useState('');
   106→
   107→  if (user) return null;
   108→
   109→  const handleLogin = async (e: React.FormEvent) => {
   110→    e.preventDefault();
   111→    setLoading(true);
   112→    try {
   113→      const fd = new FormData(e.target as HTMLFormElement);
   114→      const email = fd.get('email') as string;
   115→      const password = fd.get('password') as string;
   116→      const res = await fetch('/api/auth/login', { method: 'POST', body: JSON.stringify({ email, password }), headers: { 'Content-Type': 'application/json' } });
   117→      const data = await res.json();
   118→      if (data.success) {
   119→        setUser(data.user); addToast('Bienvenue, ' + data.user.name, 'success'); setPage('missions');
   120→      } else if (data.needs_verification) {
   121→        // Account exists but email isn't verified — switch to OTP screen.
   122→        setOtpEmail(email);
   123→        setOtpStep(true);
   124→        if (data.plain_code) {
   125→          setSimCode(data.plain_code);
   126→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   127→        } else {
   128→          addToast('Code de vérification envoyé à ' + email, 'success');
   129→        }
   130→      } else {
   131→        addToast(data.error, 'error');
   132→      }
   133→    } catch { addToast('Erreur réseau', 'error'); }
   134→    setLoading(false);
   135→  };
   136→
   137→  const handleOtpVerify = async () => {
   138→    if (!otpCode || otpCode.length < 6) { addToast('Entrez le code à 6 chiffres', 'error'); return; }
   139→    setOtpLoading(true);
   140→    try {
   141→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'verify', email: otpEmail, code: otpCode, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   142→      const data = await res.json();
   143→      if (data.success && data.user) {
   144→        setUser(data.user);
   145→        addToast('Email vérifié ! Bienvenue, ' + data.user.name, 'success');
   146→        setPage('missions');
   147→      } else {
   148→        addToast(data.error || 'Code invalide', 'error');
   149→      }
   150→    } catch { addToast('Erreur réseau', 'error'); }
   151→    setOtpLoading(false);
   152→  };
   153→
   154→  const handleResendOtp = async () => {
   155→    setOtpLoading(true);
   156→    try {
   157→      const res = await fetch('/api/auth/otp', { method: 'POST', body: JSON.stringify({ action: 'send', email: otpEmail, purpose: 'email_verification' }), headers: { 'Content-Type': 'application/json' } });
   158→      const data = await res.json();
   159→      if (data.success) {
   160→        if (data.plain_code) {
   161→          setSimCode(data.plain_code);
   162→          addToast('Code: ' + data.plain_code, 'info');
   163→        } else {
   164→          addToast('Code renvoyé à ' + otpEmail, 'success');
   165→        }
   166→      } else { addToast(data.error, 'error'); }
   167→    } catch { addToast('Erreur réseau', 'error'); }
   168→    setOtpLoading(false);
   169→  };
   170→
   171→  const handleRegister = async (e: React.FormEvent) => {
   172→    e.preventDefault();
   173→    const fd = new FormData(e.target as HTMLFormElement);
   174→    const name = (fd.get('name') as string)?.trim() || '';
   175→    const email = (fd.get('email') as string)?.trim() || '';
   176→    const phone = (fd.get('phone') as string)?.trim() || '';
   177→    const password = fd.get('password') as string || '';
   178→    const password2 = fd.get('password2') as string || '';
   179→    const referralCode = (fd.get('referralCode') as string)?.trim().toUpperCase() || '';
   180→    const errs: Record<string, string> = {};
   181→    if (name.length < 2) errs.name = 'Min. 2 caractères';
   182→    if (password.length < 6) errs.password = 'Min. 6 caractères';
   183→    if (password !== password2) errs.password2 = 'Ne correspond pas';
   184→    if (Object.keys(errs).length) { setErrors(errs); return; }
   185→    setErrors({});
   186→    setLoading(true);
   187→    try {
   188→      const res = await fetch('/api/auth/register', { method: 'POST', body: JSON.stringify({ name, email, phone: phone.trim(), password, password2, referralCode }), headers: { 'Content-Type': 'application/json' } });
   189→      const data = await res.json();
   190→      if (data.success && data.requires_verification) {
   191→        // Account created — must verify email via OTP
   192→        setOtpEmail(email);
   193→        setOtpStep(true);
   194→        if (data.plain_code) {
   195→          setSimCode(data.plain_code);
   196→          addToast('Mode simulation - Code: ' + data.plain_code, 'info');
   197→        } else {
   198→          addToast('Code de vérification envoyé à ' + email, 'success');
   199→        }
   200→      } else if (data.success) {
   201→        setUser(data.user); addToast('Compte créé !', 'success'); setPage('missions');
   202→      } else {
   203→        addToast(data.error, 'error');
   204→      }
   205→    } catch { addToast('Erreur réseau', 'error'); }
   206→    setLoading(false);
   207→  };
   208→
   209→  return (
   210→    <section className="absolute inset-0 bg-gradient-to-br from-[#F0FDF4] via-[#F8F9FA] to-[#ECFDF5] flex flex-col items-center justify-start sm:justify-center overflow-y-auto z-[200]">
   211→      {/* Floating decorative orbs */}
   212→      <div className="absolute w-[200px] h-[200px] rounded-full bg-[rgba(34,197,94,0.06)] blur-[80px] top-[15%] left-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite' }} />
   213→      <div className="absolute w-[160px] h-[160px] rounded-full bg-[rgba(20,184,166,0.05)] blur-[80px] bottom-[15%] right-[10%]" style={{ animation: 'orbFloat 6s ease-in-out infinite 3s reverse' }} />
   214→      <div className="absolute w-[120px] h-[120px] rounded-full bg-[rgba(245,158,11,0.04)] blur-[80px] top-[50%] right-[25%]" style={{ animation: 'orbFloat 7s ease-in-out infinite 1.5s' }} />
   215→
   216→      <div className="w-full max-w-[330px] text-center px-5 relative z-[1] py-6 my-auto">
   217→        <LogoImg className="w-[100px] h-[100px] mx-auto mb-4" style={{ filter: 'drop-shadow(0 8px 32px rgba(34,197,94,0.25))' }} />
   218→        <h1 className="text-[1.8rem] font-black mb-1 bg-gradient-to-r from-[#22C55E] to-[#16A34A] bg-clip-text text-transparent tracking-[2px]">JEUNE ÉLAN</h1>
   219→        {/* Platform tagline */}
   220→        <div className="inline-flex items-center gap-1.5 mb-1 px-3 py-1 rounded-full bg-[rgba(20,184,166,0.1)] border border-[rgba(20,184,166,0.2)]">
   221→          <i className="fas fa-globe-asia text-[#14B8A6] text-[0.6rem]"></i>
   222→          <span className="text-[0.58rem] font-bold text-[#0F766E] uppercase tracking-wide">Tâches rémunérées & micro-prêts</span>
   223→        </div>
   224→        {/* OTP Verification Step — Registration only */}
   225→        {otpStep ? (
   226→          <>
   227→            <div className="w-14 h-14 mx-auto rounded-full bg-[rgba(34,197,94,0.1)] flex items-center justify-center mb-4 mt-4">
   228→              <i className="fas fa-shield-alt text-[#22C55E] text-[1.3rem]"></i>
   229→            </div>
   230→            <h2 className="text-[1rem] font-bold text-[#1F2937] mb-1">Vérification email</h2>
   231→            <p className="text-[rgba(0,0,0,0.45)] text-[0.72rem] mb-5">
   232→              Un code a été envoyé à <strong className="text-[#1F2937]">{otpEmail}</strong> pour vérifier votre adresse email
   233→            </p>
   234→
   235→            {/* Simulation mode code display */}
   236→            {simCode && (
   237→              <div className="mb-4 p-4 bg-[rgba(245,158,11,0.12)] border-2 border-[rgba(245,158,11,0.4)] rounded-xl">
   238→                <div className="flex items-center gap-1.5 mb-2">
   239→                  <i className="fas fa-key text-[#F59E0B] text-[0.7rem]"></i>
   240→                  <div className="text-[0.72rem] text-[#D97706] font-bold">Votre code de vérification</div>
   241→                </div>
   242→                <div className="text-[2rem] font-black text-[#F59E0B] tracking-[8px] font-mono text-center py-2">{simCode}</div>
   243→                <div className="flex gap-2 mt-2">
   244→                  <button
   245→                    onClick={() => { setOtpCode(simCode); addToast('Code rempli automatiquement', 'success'); }}
   246→                    className="flex-1 py-2 rounded-lg bg-[#F59E0B] text-white text-[0.75rem] font-bold border-none cursor-pointer flex items-center justify-center gap-1"
   247→                  >
   248→                    <i className="fas fa-magic mr-1"></i>Remplir automatiquement
   249→                  </button>
   250→                  <button
   251→                    onClick={() => { try { navigator.clipboard.writeText(simCode); } catch { /* */ } addToast('Code copié', 'success'); }}
   252→                    className="py-2 px-3 rounded-lg bg-[rgba(245,158,11,0.15)] text-[0.72rem] text-[#D97706] font-semibold border border-[rgba(245,158,11,0.3)] cursor-pointer"

   254→                    <i className="fas fa-copy mr-1"></i>Copier
   255→                  </button>
   256→                </div>
   257→              </div>
   258→            )}
   259→
   260→            {/* 6-digit OTP input */}
   261→            <div className="flex justify-center gap-2 mb-5">
   262→              {[0, 1, 2, 3, 4, 5].map((i) => (
   263→                <input
   264→                  key={i}
   265→                  type="text"
   266→                  inputMode="numeric"
   267→                  maxLength={1}
   268→                  className="w-10 h-12 text-center text-[1.1rem] font-bold bg-white border-[1.5px] border-[rgba(0,0,0,0.1)] rounded-xl outline-none focus:border-[#22C55E] focus:shadow-[0_0_0_3px_rgba(34,197,94,0.1)] transition-all text-[#1F2937]"
   269→                  value={otpCode[i] || ''}
   270→                  onChange={(e) => {
   271→                    const val = e.target.value.replace(/\D/g, '');
   272→                    const newCode = otpCode.split('');
   273→                    newCode[i] = val.slice(-1);
   274→                    const final = newCode.join('').slice(0, 6);
   275→                    setOtpCode(final);

   277→                      const next = e.target.nextElementSibling as HTMLInputElement;
   278→                      next?.focus();
   279→                    }

   284→                      prev?.focus();
   285→                    }
   286→                  }}
   287→                  onPaste={(e) => {
   288→                    e.preventDefault();
   289→                    const paste = (e.clipboardData.getData('text') || '').replace(/\D/g, '').slice(0, 6);
   290→                    setOtpCode(paste);
   291→                  }}
   292→                />
   293→              ))}
   294→            </div>
   295→
   296→            {/* Hint when code is incomplete */}
   297→            {otpCode.length < 6 && (
   298→              <p className="text-[0.65rem] text-[rgba(0,0,0,0.4)] mb-2"><i className="fas fa-info-circle mr-1"></i>Entrez les 6 chiffres reçus par email pour finaliser</p>
   299→            )}
   300→
   301→            <button
   302→              onClick={handleOtpVerify}
   303→              disabled={otpLoading || otpCode.length < 6}
   304→              className="w-full py-3.5 rounded-xl btn-gradient-green text-[0.88rem] cursor-not-allowed disabled:opacity-60 enabled:cursor-pointer transition-transform active:scale-[0.97] flex items-center justify-center gap-2 mb-3"
   305→            >
   306→              {otpLoading ? <div className="w-4 h-4 border-2 border-[rgba(255,255,255,0.3)] border-t-white rounded-full" style={{ animation: 'spin 0.6s linear infinite' }} /> : <><i className="fas fa-check-circle"></i> Finaliser l&rsquo;inscription</>}
   307→            </button>
   308→
   309→            <button
   310→              onClick={handleResendOtp}
   311→              disabled={otpLoading}
   312→              className="w-full py-3 rounded-xl border-[1.5px] border-[rgba(0,0,0,0.08)] bg-transparent text-[rgba(0,0,0,0.5)] font-semibold text-[0.82rem] cursor-pointer transition-transform active:scale-95 mb-3"
   313→            >
   314→              <i className="fas fa-redo mr-1.5 text-[0.7rem]"></i>Renvoyer le code
   315→            </button>
   316→
   317→            <button
   318→              onClick={() => { setOtpStep(false); setOtpCode(''); setSimCode(''); }}
   319→              className="text-[0.75rem] text-[rgba(0,0,0,0.4)] cursor-pointer bg-transparent border-none font-medium"
   320→            >
   321→              <i className="fas fa-arrow-left mr-1"></i>Retour
   322→            </button>
   323→          </>
   324→        ) : (
   325→          <>
   326→            <p className="text-[rgba(0,0,0,0.35)] text-[0.72rem] mb-6">{mode === 'login' ? 'Connectez-vous à votre compte.' : 'Rejoignez Jeune Élan.'}</p>

   328→              <button onClick={() => { setMode('login'); setErrors({}); }} className={`flex-1 py-[11px] text-center text-[0.82rem] font-semibold rounded-lg transition-all border-none cursor-pointer ${mode === 'login' ? 'bg-[#22C55E] text-white shadow-lg' : 'text-[rgba(0,0,0,0.35)]'}`}>Connexion</button>
   329→              <button onClick={() => { setMode('register'); setErrors({}); }} className={`flex-1 py-[11px] text-center text-[0.82rem] font-semibold rounded-lg transition-all border-none cursor-pointer ${mode === 'register' ? 'bg-[#22C55E] text-white shadow-lg' : 'text-[rgba(0,0,0,0.35)]'}`}>Inscription</button>
   330→            </div>
   331→            {mode === 'login' ? (
   332→              <form onSubmit={handleLogin}>
   333→                {/* Email */}
   334→                <div className="mb-2 w-full relative">
   335→                  <i className="fas fa-envelope absolute left-3.5 top-1/2 -translate-y-1/2 text-[rgba(0,0,0,0.3)] text-[0.78rem] pointer-events-none"></i>
   336→                  <input name="email" type="email" required placeholder="Adresse email" className="w-full py-2.5 pl-10 pr-3.5 bg-white/80 border-[1.5px] border-[rgba(0,0,0,0.08)] rounded-xl text-[0.82rem] font-medium text-[#1F2937] placeholder:text-[rgba(0,0,0,0.35)] placeholder:font-normal outline-none focus:border-[#22C55E] focus:bg-white focus:shadow-[0_0_0_3px_rgba(34,197,94,0.1)] transition-all" />

   971→function BottomNav() {
   972→  const { currentPage, setPage } = useAppStore();
   973→  const tabs = [
   974→    { id: 'missions', icon: 'fa-th-large', label: 'Tableau' },
   975→    { id: 'home', icon: 'fa-home', label: 'Accueil' },
   976→    { id: 'finance', icon: 'fa-chart-line', label: 'Finance' },
   977→    { id: 'wallet', icon: 'fa-wallet', label: 'Portefeuille' },
   978→    { id: 'guide', icon: 'fa-compass', label: 'Guide' },
   979→    { id: 'profile', icon: 'fa-user', label: 'Profil' },
   980→  ];
   981→  const isActive = (tabId: string) => {
   982→    if (tabId === 'missions') return ['missions'].includes(currentPage);
   983→    if (tabId === 'home') return currentPage === 'home';
   984→    if (tabId === 'finance') return ['finance', 'invest', 'game', 'enterprise', 'deposit', 'deposit-choose', 'withdraw'].includes(currentPage);
   985→    if (tabId === 'wallet') return ['wallet'].includes(currentPage);
   986→    if (tabId === 'guide') return currentPage === 'guide';
   987→    if (tabId === 'profile') return ['profile', 'analytics', 'admin', 'chat'].includes(currentPage);
   988→    return currentPage === tabId;
   989→  };
   990→  return (
   991→    <nav className="h-[60px] bg-white/90 backdrop-blur-xl border-t border-[rgba(0,0,0,0.06)] flex items-center justify-around px-0.5 shrink-0 safe-area-bottom">
   992→      {tabs.map(t => (
   993→        <button key={t.id} onClick={() => setPage(t.id)} className={`flex flex-col items-center justify-center py-1.5 px-1 border-none cursor-pointer transition-all relative ${isActive(t.id) ? 'text-[#22C55E]' : 'text-[rgba(0,0,0,0.3)]'}`}>
   994→          {isActive(t.id) && <div className="absolute -top-0.5 w-5 h-[3px] rounded-full bg-[#22C55E]"></div>}
   995→          <i className={`fas ${t.icon} text-[0.85rem] mb-0.5`}></i>
   996→          <span className={`text-[0.48rem] ${isActive(t.id) ? 'font-bold text-[#22C55E]' : 'font-semibold'}`}>{t.label}</span>
   997→        </button>
   998→      ))}
   999→    </nav>
  1000→  );
