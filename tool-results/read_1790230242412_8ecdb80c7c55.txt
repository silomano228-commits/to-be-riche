   348→function DashV({ firstName, setPage, go }: { firstName: string; setPage: (p: string) => void; go: (t: TabId) => void }) {
   349→  const s = MOCK_STATS;
   350→
   351→  /* Valeurs calculées — resteront exactes quand les vraies données
   352→     remplaceront les données fictives. IMPORTANT (spec §11) : la
   353→     progression vers 2 500 F se base sur les gains issus des missions
   354→     validées, PAS sur le solde (dépôts, caution et prêts exclus). */
   355→  const pctObjective = Math.min(100, Math.round((s.missionGainsCfa / s.objectiveCfa) * 100));
   356→  const restObjective = Math.max(0, s.objectiveCfa - s.missionGainsCfa);
   357→  const pctSubs = Math.min(100, Math.round((s.todaySubmissions / s.dailyLimit) * 100));
   358→  const restSubs = Math.max(0, s.dailyLimit - s.todaySubmissions);
   359→  const pctRef = Math.min(100, Math.round((s.referralCount / s.referralRequired) * 100));
   360→  const restRef = Math.max(0, s.referralRequired - s.referralCount);
   361→  const cautionReady = s.cautionBalanceCfa >= s.cautionRequiredCfa;
   362→  const gainsOk = s.missionGainsCfa >= s.objectiveCfa;
   363→  const referralsOk = s.referralCount >= s.referralRequired;
   364→  const eligible = cautionReady && gainsOk && referralsOk && s.accountVerified;
   365→  const missing: string[] = [];
   366→  if (!gainsOk) missing.push(`${formatCfa(restObjective)} de gains`);
   367→  if (!cautionReady) missing.push('la caution');
   368→  if (!referralsOk) missing.push(`${restRef} parrainage${restRef > 1 ? 's' : ''}`);
   369→
